   886→// === BROADCAST STATS DIALOG (queue progress + anti-banned tracking) ===
   887→function BroadcastStatsDialog({ broadcast, onClose }: { broadcast: any; onClose: () => void }) {
   888→  const [stats, setStats] = useState<any>(null)
   889→  const [loading, setLoading] = useState(true)
   890→  const [processing, setProcessing] = useState(false)
   891→
   892→  const loadData = useCallback(async () => {
   893→    try {
   894→      const res = await fetch(`/api/broadcast-composer/${broadcast.id}/stats`, {
   895→        headers: { 'x-user-id': useAuthStore.getState().user?.id || '' },
   896→      })
   897→      const data = await res.json()
   898→      if (data.success) setStats(data.data)
   899→    } catch (e) { /* ignore */ }
   900→    finally { setLoading(false) }
   901→  }, [broadcast.id])
   902→
   903→  useEffect(() => {
   904→    loadData()
   905→    const interval = setInterval(loadData, 15000) // 15s (was 5s — too aggressive for Vercel serverless)
   906→    return () => clearInterval(interval)
   907→  }, [loadData])
   908→
   909→  const handleProcessQueue = async () => {
   910→    setProcessing(true)
   911→    try {
   912→      const res = await fetch(`/api/broadcast-composer/${broadcast.id}/queue`, {
   913→        method: 'POST',
   914→        headers: { 'x-user-id': useAuthStore.getState().user?.id || '' },
   915→      })
   916→      const data = await res.json()
   917→      if (data.success) {
   918→        useToastStore.getState().addToast(data.message, 'success')
   919→        loadData()
   920→      }
   921→    } catch (e: any) { useToastStore.getState().addToast(e.message, 'error') }
   922→    finally { setProcessing(false) }
   923→  }
   924→
   925→  if (loading) return <LoadingState />
   926→
   927→  return (
   928→    <Dialog open={true} onOpenChange={onClose}>
   929→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto" aria-describedby={undefined}>
   930→        <DialogHeader>
   931→          <DialogTitle className="flex items-center gap-2"><Activity className="w-5 h-5 text-orange-600" /> Broadcast Stats: {broadcast.title}</DialogTitle>
   932→          <DialogDescription>Progress pengiriman + anti-banned queue tracking</DialogDescription>
   933→        </DialogHeader>
   934→
   935→        {stats && (
   936→          <div className="space-y-3">
   937→            {/* Progress bar */}
   938→            <div className="rounded-lg bg-slate-50 border p-3">
   939→              <div className="flex items-center justify-between mb-2">
   940→                <span className="text-sm font-semibold">Progress Pengiriman</span>
   941→                <span className="text-2xl font-bold text-orange-600">{stats.stats.progress}%</span>
   942→              </div>
   943→              <div className="w-full bg-slate-200 rounded-full h-3 overflow-hidden">
   944→                <div className="bg-gradient-to-r from-orange-500 to-red-600 h-full transition-all" style={{ width: `${stats.stats.progress}%` }} />
   945→              </div>
   946→              <div className="grid grid-cols-4 gap-2 mt-3 text-center text-xs">
   947→                <div className="rounded bg-amber-50 p-1.5"><div className="font-bold text-amber-700">{stats.stats.queued}</div><div className="text-[13px]">Antrian</div></div>
   948→                <div className="rounded bg-emerald-50 p-1.5"><div className="font-bold text-emerald-700">{stats.stats.sent}</div><div className="text-[13px]">Terkirim</div></div>
   949→                <div className="rounded bg-red-50 p-1.5"><div className="font-bold text-red-700">{stats.stats.failed}</div><div className="text-[13px]">Gagal</div></div>
   950→                <div className="rounded bg-slate-100 p-1.5"><div className="font-bold text-slate-700">{stats.stats.blocked}</div><div className="text-[13px]">Blocked</div></div>
   951→              </div>
   952→              <div className="text-[13px] text-muted-foreground mt-2 text-center">
   953→                Total: {stats.stats.total} • Success rate: {stats.stats.successRate}%
   954→              </div>
   955→            </div>
   956→

   962→              </Button>

   964→
   965→            {/* Sample sent messages (personalized) */}
   966→            {stats.sentMessages && stats.sentMessages.length > 0 && (
   967→              <Card>
   968→                <CardHeader><CardTitle className="text-sm">Sample Pesan Terkirim (Personalisasi Aktif)</CardTitle></CardHeader>
