   132→export function PusatAdminMenu() {
   133→  const user = useAuthStore((s) => s.user)
   134→  const setNav = useNavStore((s) => s.setActiveMenu)
   135→  const [stats, setStats] = useState<StatsData | null>(null)
   136→  const [loadingStats, setLoadingStats] = useState(true)
   137→  const [statsError, setStatsError] = useState<string | null>(null)
   138→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null)
   139→
   140→  // Load stats
   141→  const loadStats = useCallback(async () => {
   142→    setLoadingStats(true)
   143→    setStatsError(null)
   144→    try {
   145→      const data = await api('/api/stats')
   146→      setStats(data as StatsData)
   147→      setLastUpdated(new Date())
   148→    } catch (e: any) {
   149→      console.error('[PusatAdmin] stats load failed:', e)
   150→      setStatsError(e.message || 'Gagal memuat statistik')
   151→    } finally {
   152→      setLoadingStats(false)
   153→    }
   154→  }, [])
   155→
   156→  useEffect(() => { loadStats() }, [loadStats])
   157→
   158→  return (
   159→    <div className="space-y-6">
   160→      {/* Header dengan tombol refresh + timestamp */}
   161→      <div className="flex items-start justify-between gap-4 flex-wrap">
   162→        <PageHeader
   163→          title="Pusat Admin"
   164→          description="Ringkasan operasional & manajemen pengguna sistem"
   165→          icon={ShieldCheck}
   166→        />
   167→        <div className="flex items-center gap-2">
   168→          {lastUpdated && (
   169→            <span className="text-xs text-muted-foreground">
   170→              Diperbarui: {lastUpdated.toLocaleTimeString('id-ID')}
   171→            </span>
   172→          )}
   173→          <Button variant="outline" size="sm" onClick={loadStats} disabled={loadingStats} className="gap-1">
   174→            <RefreshCw className={`w-3 h-3 ${loadingStats ? 'animate-spin' : ''}`} /> Refresh
   175→          </Button>
   176→        </div>
   177→      </div>
   178→
   179→      {/* KPI Cards (4) — ringkasan kilat */}
   180→      {loadingStats ? (
   181→        <LoadingState message="Memuat ringkasan..." />
   182→      ) : statsError ? (
   183→        <ErrorState message={statsError} />
   184→      ) : stats ? (
   185→        <>
   186→          <KpiCards stats={stats} />
   187→          <AlertCallout stats={stats} onNavigate={setNav} />
   188→          <QuickActions onNavigate={setNav} />
   189→        </>
   190→      ) : null}
   191→
   192→      {/* Manajemen User — inline, bukan tab */}
   193→      <UsersManager />
   194→
   195→      {/* Saklar Keamanan — SUPERADMIN only, accordion */}
   196→      {user?.role === 'SUPERADMIN' && <SecurityManager />}
   197→    </div>
   198→  )
   199→}

   256→// Helper format IDR singkat (1.2M, 500K, dll)
   257→function formatIDRShort(value: number): string {
   258→  if (!value) return 'Rp 0'
   259→  const abs = Math.abs(value)
   260→  if (abs >= 1_000_000_000) return `Rp ${(value / 1_000_000_000).toFixed(1)}M`
   261→  if (abs >= 1_000_000) return `Rp ${(value / 1_000_000).toFixed(1)}jt`
   262→  if (abs >= 1_000) return `Rp ${(value / 1_000).toFixed(0)}K`
   263→  return `Rp ${value}`
   264→}

   269→function AlertCallout({ stats, onNavigate }: { stats: StatsData; onNavigate: (menu: string) => void }) {

   273→

   275→    <Card className="border-orange-300 bg-orange-50">
   276→      <CardContent className="p-3 flex items-center gap-3 flex-wrap">
   277→        <AlertTriangle className="w-5 h-5 text-orange-600 shrink-0" />
   278→        <div className="flex-1 min-w-0">
   279→          <div className="text-sm font-medium text-orange-900">
   280→            {pending > 0 && `${pending} anggota menunggu verifikasi`}
   281→            {pending > 0 && rejected > 0 && ' · '}
   282→            {rejected > 0 && `${rejected} anggota ditolak`}
   283→          </div>
   284→          <div className="text-xs text-orange-700">Klik untuk membuka menu Keanggotaan</div>
   285→        </div>
   286→        <Button size="sm" variant="outline" onClick={() => onNavigate('membership')} className="gap-1">
   287→          Buka <ArrowRight className="w-3 h-3" />
   288→        </Button>

   292→}

   324→// USERS MANAGER — inline dengan search + filter + pagination + action
   325→// ============================================================
   326→function UsersManager() {
   327→  const addToast = useToastStore((s) => s.addToast)
   328→  const user = useAuthStore((s) => s.user)
   329→  const [users, setUsers] = useState<UserItem[]>([])
   330→  const [territories, setTerritories] = useState<Territory[]>([])
   331→  const [loading, setLoading] = useState(true)
   332→  const [error, setError] = useState<string | null>(null)
   333→  const [search, setSearch] = useState('')
   334→  const [searchDebounced, setSearchDebounced] = useState('')
   335→  const [roleFilter, setRoleFilter] = useState<string>('ALL')
   336→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   337→  const [page, setPage] = useState(1)
   338→  const [addOpen, setAddOpen] = useState(false)
   339→  const [editUser, setEditUser] = useState<UserItem | null>(null)
   340→  const [deleteUser, setDeleteUser] = useState<UserItem | null>(null)
   341→  const [resetPasswordUser, setResetPasswordUser] = useState<UserItem | null>(null)
   342→
   343→  // Debounce search
   344→  useEffect(() => {
   345→    const t = setTimeout(() => {
   346→      setSearchDebounced(search)
   347→      setPage(1)
   348→    }, 250)
   349→    return () => clearTimeout(t)
   350→  }, [search])
   351→
   352→  // Load users + territories
   353→  const loadData = useCallback(async () => {
   354→    setLoading(true)
   355→    setError(null)
   356→    try {
   357→      const [u, t] = await Promise.all([
   358→        api('/api/users').catch(() => []),
   359→        api('/api/territory').catch(() => []),
   360→      ])
   361→      setUsers(u as UserItem[])
   362→      setTerritories(t as Territory[])
   363→    } catch (e: any) {
   364→      setError(e.message || 'Gagal memuat data user')
   365→    } finally {
   366→      setLoading(false)
   367→    }
   368→  }, [])
   369→
   370→  useEffect(() => { loadData() }, [loadData])
   371→
   372→  // Filtered + paginated users
   373→  const filtered = useMemo(() => {
   374→    let result = users
   375→    if (searchDebounced) {
   376→      const q = searchDebounced.toLowerCase()
   377→      result = result.filter(u =>
   378→        u.username.toLowerCase().includes(q) ||
   379→        u.fullName.toLowerCase().includes(q) ||
   380→        u.territory?.name?.toLowerCase().includes(q)
   381→      )
   382→    }
   383→    if (roleFilter !== 'ALL') {
   384→      result = result.filter(u => u.role === roleFilter)
   385→    }
   386→    if (statusFilter !== 'ALL') {
   387→      const wantActive = statusFilter === 'ACTIVE'
   388→      result = result.filter(u => u.isActive === wantActive)
   389→    }
   390→    return result
   391→  }, [users, searchDebounced, roleFilter, statusFilter])
   392→
   393→  const paginated = useMemo(() => {
   394→    const start = (page - 1) * PAGE_SIZE
   395→    return filtered.slice(start, start + PAGE_SIZE)
   396→  }, [filtered, page])
   397→
   398→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE))
   399→
   400→  // Cek role boleh tambah user
   401→  const canAddUser = user?.role === 'SUPERADMIN' || user?.role === 'ADMIN_DPN' || user?.role === 'ADMIN_DPD'
   402→  const canEditUser = (target: UserItem) => {
   403→    if (!user) return false
   404→    if (user.id === target.id) return true // self edit
   405→    if (user.role === 'SUPERADMIN' || user.role === 'ADMIN_DPN') return true
   406→    if (user.role === 'ADMIN_DPD' && target.role === 'ADMIN_DPC') return true
   407→    return false
   408→  }
   409→  const canDeleteUser = user?.role === 'SUPERADMIN' || user?.role === 'ADMIN_DPN'
   410→
   411→  return (
   412→    <Card>
   413→      <CardHeader className="pb-3">
   414→        <div className="flex items-center justify-between gap-2 flex-wrap">
   415→          <div>
   416→            <CardTitle className="text-base flex items-center gap-2">
   417→              <UserCog className="w-4 h-4" /> Manajemen User
   418→            </CardTitle>
   419→            <CardDescription className="text-xs">
   420→              {filtered.length} user{filtered.length !== users.length && ` dari ${users.length} total`}
   421→            </CardDescription>
   422→          </div>
   423→          {canAddUser && (
   424→            <Button size="sm" onClick={() => setAddOpen(true)} className="gap-1">
   425→              <Plus className="w-4 h-4" /> Tambah User
   426→            </Button>
   427→          )}
   428→        </div>
   429→      </CardHeader>
   430→      <CardContent className="space-y-3">
   431→        {/* Filter bar */}
   432→        <div className="flex flex-wrap gap-2">
   433→          <div className="relative flex-1 min-w-48">
   434→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   435→            <Input
   436→              placeholder="Cari username / nama / wilayah..."
   437→              value={search}
   438→              onChange={(e) => setSearch(e.target.value)}
   439→              className="pl-9 h-9"
   440→            />
   441→            {search && (
   442→              <button
   443→                type="button"
   444→                onClick={() => setSearch('')}
   445→                className="absolute right-2 top-1/2 -translate-y-1/2 p-1 text-muted-foreground hover:text-foreground"
   446→              >
   447→                <XCircle className="w-3 h-3" />
   448→              </button>
   449→            )}
   450→          </div>
   451→          <Select value={roleFilter} onValueChange={(v) => { setRoleFilter(v); setPage(1) }}>
   452→            <SelectTrigger className="w-36 h-9"><SelectValue placeholder="Role" /></SelectTrigger>
   453→            <SelectContent>
   454→              <SelectItem value="ALL">Semua Role</SelectItem>
   455→              <SelectItem value="SUPERADMIN">Super Admin</SelectItem>
   456→              <SelectItem value="ADMIN_DPN">Admin DPN</SelectItem>
   457→              <SelectItem value="ADMIN_DPD">Admin DPD</SelectItem>
   458→              <SelectItem value="ADMIN_DPC">Admin DPC</SelectItem>
   459→            </SelectContent>
   460→          </Select>
   461→          <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1) }}>
   462→            <SelectTrigger className="w-32 h-9"><SelectValue placeholder="Status" /></SelectTrigger>
   463→            <SelectContent>
   464→              <SelectItem value="ALL">Semua</SelectItem>
   465→              <SelectItem value="ACTIVE">Aktif</SelectItem>
   466→              <SelectItem value="INACTIVE">Nonaktif</SelectItem>
   467→            </SelectContent>
   468→          </Select>
   469→        </div>
   470→
   471→        {/* Error state */}
   472→        {error && <ErrorState message={error} />}
   473→
   474→        {/* Loading */}
   475→        {loading && !error && <LoadingState message="Memuat user..." />}
   476→
   477→        {/* Empty state */}
   478→        {!loading && !error && paginated.length === 0 && (
   479→          <EmptyState
   480→            icon={UserCog}
   481→            title={search || roleFilter !== 'ALL' || statusFilter !== 'ALL' ? 'Tidak ada user cocok' : 'Belum ada user'}
   482→            description={search || roleFilter !== 'ALL' || statusFilter !== 'ALL' ? 'Coba ubah filter' : 'Tambah user baru untuk mulai.'}
   483→          />
   484→        )}
   485→
   486→        {/* Tabel user */}
   487→        {!loading && !error && paginated.length > 0 && (
   488→          <div className="rounded-lg border overflow-x-auto">
   489→            <Table>
   490→              <TableHeader>
   491→                <TableRow>
   492→                  <TableHead className="w-32">Username</TableHead>
   493→                  <TableHead>Nama</TableHead>
   494→                  <TableHead className="w-28">Role</TableHead>
   495→                  <TableHead className="w-40">Wilayah</TableHead>
   496→                  <TableHead className="w-20">Status</TableHead>
   497→                  <TableHead className="w-32">Login Terakhir</TableHead>
   498→                  <TableHead className="w-28 text-right">Aksi</TableHead>
   499→                </TableRow>
   500→              </TableHeader>
   501→              <TableBody>
   502→                {paginated.map(u => (
   503→                  <TableRow key={u.id} className={!u.isActive ? 'opacity-60' : ''}>
   504→                    <TableCell className="font-mono text-xs">{u.username}</TableCell>
   505→                    <TableCell className="font-medium text-sm">{u.fullName}</TableCell>
   506→                    <TableCell>
   507→                      <Badge variant="outline" className={`text-[10px] ${ROLE_COLORS[u.role] || ''}`}>
   508→                        {ROLE_LABELS[u.role] || u.role}
   509→                      </Badge>
   510→                    </TableCell>
   511→                    <TableCell className="text-xs">{u.territory?.name || '-'}</TableCell>
   512→                    <TableCell>
   513→                      {u.isActive ? (
   514→                        <span className="flex items-center gap-1 text-xs text-green-600">
   515→                          <CheckCircle2 className="w-3 h-3" /> Aktif
   516→                        </span>
   517→                      ) : (
   518→                        <span className="flex items-center gap-1 text-xs text-red-600">
   519→                          <XCircle className="w-3 h-3" /> Nonaktif
   520→                        </span>
   521→                      )}
   522→                    </TableCell>
   523→                    <TableCell className="text-xs text-muted-foreground">
   524→                      {u.lastLogin ? formatDateTimeID(u.lastLogin) : 'Belum pernah'}
   525→                    </TableCell>
   526→                    <TableCell>

   973→function ResetPasswordDialog({
   974→  user, open, onOpenChange, onSuccess,

   978→  onOpenChange: (o: boolean) => void
   979→  onSuccess: () => void
   980→}) {

  1009→      onSuccess()
  1010→    } catch (e: any) {
  1011→      addToast(`Gagal reset password: ${e.message}`, 'error')
  1012→    } finally {
  1013→      setSaving(false)
  1014→    }
  1015→  }

  1018→    <Dialog open={open} onOpenChange={onOpenChange}>

  1026→            <Label>Password Baru *</Label>
  1027→            <Input
  1028→              type="password"
  1029→              value={newPassword}
  1030→              onChange={(e) => setNewPassword(e.target.value)}
  1031→              placeholder="Min. 6 karakter"
  1032→            />
  1033→          </div>
  1034→          <div className="space-y-1.5">
  1035→            <Label>Konfirmasi Password *</Label>

  1047→            Reset Password
  1048→          </Button>
  1049→        </DialogFooter>
  1050→      </DialogContent>

  1070→    } finally {
  1071→      setLoading(false)

  1076→

  1078→    try {
  1079→      await api('/api/security', {
  1080→        method: 'PATCH',
  1081→        body: JSON.stringify({ key: setting.key, isActive: !setting.isActive }),
