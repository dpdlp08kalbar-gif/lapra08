  1251→function DocumentUploadDialog({
  1252→  open, onOpenChange, type, onUploaded, addToast,
  1253→}: {
  1254→  open: boolean
  1255→  onOpenChange: (v: boolean) => void
  1256→  type: 'AD_ART' | 'LEGALITAS'
  1257→  onUploaded: () => void
  1258→  addToast: (msg: string, type?: 'success' | 'error' | 'info' | 'warning') => void
  1259→}) {
  1260→  const [title, setTitle] = useState('')
  1261→  const [description, setDescription] = useState('')
  1262→  const [file, setFile] = useState<File | null>(null)
  1263→  const [saving, setSaving] = useState(false)
  1264→
  1265→  useEffect(() => {
  1266→    if (open) {
  1267→      setTitle(''); setDescription(''); setFile(null)
  1268→    }
  1269→  }, [open])
  1270→
  1271→  const handleSubmit = async () => {
  1272→    if (!file) {
  1273→      addToast('Pilih file terlebih dahulu', 'warning')
  1274→      return
  1275→    }
  1276→    if (!title.trim()) {
  1277→      addToast('Judul wajib diisi', 'warning')
  1278→      return
  1279→    }
  1280→    setSaving(true)
  1281→    try {
  1282→      const fd = new FormData()
  1283→      fd.append('file', file)
  1284→      fd.append('title', title.trim())
  1285→      fd.append('description', description.trim())
  1286→      fd.append('docType', type)
  1287→      const res = await fetch('/api/profile-documents', {
  1288→        method: 'POST',
  1289→        headers: { 'x-user-id': useAuthStore.getState().user?.id || '' },
  1290→        body: fd,
  1291→      })
  1292→      const data = await res.json()
  1293→      if (!data.success) throw new Error(data.error || 'Gagal upload')
  1294→      addToast('Dokumen berhasil diupload', 'success')
  1295→      onUploaded()
  1296→    } catch (e: any) {
  1297→      addToast(e.message || 'Gagal upload dokumen', 'error')
  1298→    } finally {
  1299→      setSaving(false)
  1300→    }
  1301→  }
  1302→
  1303→  return (
  1304→    <Dialog open={open} onOpenChange={onOpenChange}>
  1305→      <DialogContent className="max-w-lg" aria-describedby={undefined}>
  1306→        <DialogHeader>
  1307→          <DialogTitle>Upload Dokumen {type === 'AD_ART' ? 'AD/ART' : 'Legalitas'}</DialogTitle>
  1308→          <DialogDescription>
  1309→            Format yang didukung: PDF, JPG, PNG, DOC/DOCX. Maksimal 20MB.
  1310→          </DialogDescription>
  1311→        </DialogHeader>
  1312→        <div className="space-y-4 py-2">
  1313→          <div className="space-y-1">
  1314→            <Label className="text-xs">Judul Dokumen *</Label>
  1315→            <Input

  1317→              onChange={(e) => setTitle(e.target.value)}
  1318→              placeholder="cth: AD/ART LAPRA 08 Periode 2024-2029"
  1319→            />
  1320→          </div>
  1321→          <div className="space-y-1">
  1322→            <Label className="text-xs">Deskripsi (opsional)</Label>
  1323→            <Textarea
  1324→              value={description}
  1325→              onChange={(e) => setDescription(e.target.value)}
  1326→              rows={2}
  1327→              placeholder="Ringkasan singkat dokumen..."
  1328→            />
  1329→          </div>
  1330→          <div className="space-y-1">
  1331→            <Label className="text-xs">File Dokumen *</Label>
  1332→            <Input
  1333→              type="file"
