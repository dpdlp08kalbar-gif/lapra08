   218→// HALAMAN DPD LIST - Daftar semua DPD
   219→// ============================================================
   220→function DpdListPage({ onBack, onSelectDpd }: { onBack: () => void; onSelectDpd: (t: Territory) => void }) {
   221→  const user = useAuthStore((s) => s.user)!
   222→  const addToast = useToastStore((s) => s.addToast)
   223→  const [dpds, setDpds] = useState<Territory[]>([])
   224→  const [loading, setLoading] = useState(true)
   225→  const [error, setError] = useState('')
   226→  const [search, setSearch] = useState('')
   227→  const [addOpen, setAddOpen] = useState(false)
   228→  const [editDpd, setEditDpd] = useState<Territory | null>(null)
   229→  const [deleteDpd, setDeleteDpd] = useState<Territory | null>(null)
   230→
   231→  const loadData = () => {
   232→    setLoading(true)
   233→    api('/api/territory?level=PROVINCE')
   234→      .then(setDpds)
   235→      .catch((e) => setError(e.message))
   236→      .finally(() => setLoading(false))
   237→  }
   238→  useEffect(() => { loadData() }, [])
   239→
   240→  const handleDelete = async () => {
   241→    if (!deleteDpd) return
   242→    try {
   243→      await api(`/api/territory/${deleteDpd.id}`, { method: 'DELETE' })
   244→      addToast(`DPD "${deleteDpd.name}" berhasil dihapus`, 'success')
   245→      setDeleteDpd(null); loadData()
   246→    } catch (e: any) { addToast(e.message, 'error') }
   247→  }
   248→
   249→  if (loading) return <><BackButton onBack={onBack} label="DPD (Provinsi)" /><LoadingState /></>
   250→  if (error) return <><BackButton onBack={onBack} label="DPD (Provinsi)" /><ErrorState message={error} /></>
   251→
   252→  const filtered = dpds.filter((d) =>
   253→    !search || d.name.toLowerCase().includes(search.toLowerCase()) || d.code.toLowerCase().includes(search.toLowerCase())
   254→  )
   255→  const domestic = filtered.filter((d) => d.category === 'DOMESTIC')
   256→  const intl = filtered.filter((d) => d.category === 'INTERNATIONAL')
   257→
   258→  const canManage = user.role === 'SUPERADMIN' || user.role === 'ADMIN_DPN'
   259→
   260→  return (
   261→    <div className="space-y-4">
   262→      <BackButton onBack={onBack} label="DPD (Provinsi)" />
   263→
   264→      {/* Toolbar */}
   265→      <div className="flex flex-col sm:flex-row gap-2 justify-between">
   266→        <div className="relative flex-1 max-w-md">
   267→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   268→          <Input placeholder="Cari DPD..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-10" />
   269→        </div>
   270→        {canManage && (
   271→          <Button onClick={() => setAddOpen(true)} className="bg-gradient-to-r from-orange-600 to-red-600 text-white">
   272→            <Plus className="w-4 h-4 mr-2" /> Tambah DPD
   273→          </Button>
   274→        )}
   275→      </div>
   276→
   277→      {/* DPD Domestik */}
   278→      <Card>
   279→        <CardHeader>
   280→          <CardTitle className="flex items-center gap-2 text-base">
   281→            <Building className="w-4 h-4 text-blue-600" />
   282→            DPD Domestik ({domestic.length})
   283→          </CardTitle>
   284→        </CardHeader>
   285→        <CardContent>
   286→          <div className="grid gap-2 md:grid-cols-2 lg:grid-cols-3">
   287→            {domestic.map((d) => (
   288→              <TerritoryCard key={d.id} territory={d} onClick={() => onSelectDpd(d)} canManage={canManage} onEdit={() => setEditDpd(d)} onDelete={() => setDeleteDpd(d)} />
   289→            ))}
   290→          </div>
   291→        </CardContent>
   292→      </Card>
   293→
   294→      {/* DPD Luar Negeri */}
   295→      {intl.length > 0 && (
   296→        <Card>
   297→          <CardHeader>
   298→            <CardTitle className="flex items-center gap-2 text-base">
   299→              <Building className="w-4 h-4 text-pink-600" />
   300→              DPD Luar Negeri ({intl.length})
   301→            </CardTitle>
   302→          </CardHeader>
   303→          <CardContent>
   304→            <div className="grid gap-2 md:grid-cols-2 lg:grid-cols-3">
   305→              {intl.map((d) => (
   306→                <TerritoryCard key={d.id} territory={d} onClick={() => onSelectDpd(d)} canManage={canManage} onEdit={() => setEditDpd(d)} onDelete={() => setDeleteDpd(d)} />
   307→              ))}
   308→            </div>
   309→          </CardContent>
   310→        </Card>
   311→      )}
   312→
   313→      {/* Dialogs */}
   314→      <AddTerritoryDialog open={addOpen} onOpenChange={setAddOpen} level="PROVINCE" onSuccess={loadData} />
   315→      <EditTerritoryDialog territory={editDpd} onOpenChange={(o) => !o && setEditDpd(null)} onSuccess={() => { loadData(); setEditDpd(null); addToast('DPD diperbarui', 'success') }} />
   316→      <AlertDialog open={!!deleteDpd} onOpenChange={(o) => !o && setDeleteDpd(null)}>
   317→        <AlertDialogContent>
   318→          <AlertDialogHeader>
   319→            <AlertDialogTitle>Hapus DPD?</AlertDialogTitle>
   320→            <AlertDialogDescription>
   321→              Yakin hapus <strong>{deleteDpd?.name}</strong> ({deleteDpd?.code})?
   322→              DPD yang masih memiliki DPC atau anggota tidak bisa dihapus.
   323→            </AlertDialogDescription>
   324→          </AlertDialogHeader>
   325→          <AlertDialogFooter>
   326→            <AlertDialogCancel>Batal</AlertDialogCancel>
   327→            <AlertDialogAction onClick={handleDelete} className="bg-red-600 hover:bg-red-700">Hapus</AlertDialogAction>
   328→          </AlertDialogFooter>
   329→        </AlertDialogContent>
   330→      </AlertDialog>
   331→    </div>
   332→  )
   333→}

   382→function DpcListPage({ onBack, onSelectDpc }: { onBack: () => void; onSelectDpc: (t: Territory) => void }) {
   383→  const user = useAuthStore((s) => s.user)!
   384→  const addToast = useToastStore((s) => s.addToast)
   385→  const [dpds, setDpds] = useState<Territory[]>([])
   386→  const [selectedDpdId, setSelectedDpdId] = useState<string>('')
   387→  const [dpcs, setDpcs] = useState<Territory[]>([])
   388→  const [loading, setLoading] = useState(true)
   389→  const [addOpen, setAddOpen] = useState(false)
   390→  const [editDpc, setEditDpc] = useState<Territory | null>(null)
   391→  const [deleteDpc, setDeleteDpc] = useState<Territory | null>(null)
   392→
   393→  const loadDpds = () => {
   394→    api('/api/territory?level=PROVINCE').then((d) => {
   395→      setDpds(d)
   396→      if (user.role === 'ADMIN_DPD') {
   397→        setSelectedDpdId(user.territoryId)
   398→      } else if (d.length > 0) {
   399→        const kalbar = d.find((t: Territory) => t.code === '61')
   400→        setSelectedDpdId(kalbar?.id || d[0].id)
   401→      }
   402→      setLoading(false)
   403→    }).catch(() => setLoading(false))
   404→  }
   405→
   406→  useEffect(() => { loadDpds() }, [])
   407→
   408→  const loadDpcs = () => {
   409→    if (selectedDpdId) {
   410→      api(`/api/territory?parentId=${selectedDpdId}`).then(setDpcs).catch(() => setDpcs([]))
   411→    } else {
   412→      setDpcs([])
   413→    }
   414→  }
   415→
   416→  useEffect(() => { loadDpcs() }, [selectedDpdId])
   417→
   418→  const handleDelete = async () => {
   419→    if (!deleteDpc) return
   420→    try {
   421→      await api(`/api/territory/${deleteDpc.id}`, { method: 'DELETE' })
   422→      addToast(`DPC "${deleteDpc.name}" berhasil dihapus`, 'success')
   423→      setDeleteDpc(null); loadDpcs()
   424→    } catch (e: any) { addToast(e.message, 'error') }
   425→  }
   426→
   427→  if (loading) return <><BackButton onBack={onBack} label="DPC (Kabupaten/Kota)" /><LoadingState /></>
   428→
   429→  const canManage = user.role === 'SUPERADMIN' || user.role === 'ADMIN_DPN' || user.role === 'ADMIN_DPD'
   430→
   431→  return (
   432→    <div className="space-y-4">
   433→      <BackButton onBack={onBack} label="DPC (Kabupaten/Kota)" />
   434→
   435→      {/* Filter DPD */}
   436→      <div className="space-y-2">
   437→        <Label>Pilih DPD (Provinsi) untuk melihat DPC</Label>
   438→        <Select value={selectedDpdId} onValueChange={setSelectedDpdId}>
   439→          <SelectTrigger className="w-full max-w-md"><SelectValue placeholder="Pilih DPD..." /></SelectTrigger>
   440→          <SelectContent>
   441→            {dpds.map((d) => (
   442→              <SelectItem key={d.id} value={d.id}>{d.name} ({d.code})</SelectItem>
   443→            ))}
   444→          </SelectContent>
   445→        </Select>
   446→      </div>
   447→
   448→      {selectedDpdId && (
   449→        <Card>
   450→          <CardHeader>
   451→            <div className="flex items-center justify-between flex-wrap gap-2">
   452→              <CardTitle className="flex items-center gap-2 text-base">
   453→                <MapPin className="w-4 h-4 text-emerald-600" />
   454→                Daftar DPC ({dpcs.length})
   455→              </CardTitle>
   456→              {canManage && (
   457→                <Button onClick={() => setAddOpen(true)} size="sm" className="bg-gradient-to-r from-orange-600 to-red-600 text-white">
   458→                  <Plus className="w-4 h-4 mr-1" /> Tambah DPC
   459→                </Button>
   460→              )}
   461→            </div>
   462→          </CardHeader>
   463→          <CardContent>

   534→// ============================================================
   535→// SECTION PENGURUS (Reusable untuk DPN/DPD/DPC)
   536→// ============================================================
   537→function PengurusSection({ level, territoryId, territoryFilter }: {
   538→  level: string; territoryId?: string; territoryFilter?: { level: string }
   539→}) {
   540→  const user = useAuthStore((s) => s.user)!
   541→  const addToast = useToastStore((s) => s.addToast)
   542→  const [positions, setPositions] = useState<OrgPosition[]>([])
   543→  const [loading, setLoading] = useState(true)
   544→  const [error, setError] = useState('')
   545→  const [addOpen, setAddOpen] = useState(false)
   546→  const [editPos, setEditPos] = useState<OrgPosition | null>(null)
   547→  const [deletePos, setDeletePos] = useState<OrgPosition | null>(null)
   548→
   549→  const loadData = () => {
   550→    setLoading(true)
   551→    api('/api/organization').then((all) => {
   552→      let filtered = all.filter((p: OrgPosition) => p.level === level)
   553→      if (territoryId) {
   554→        filtered = filtered.filter((p: OrgPosition) => p.territoryId === territoryId)
   555→      } else if (territoryFilter) {
   556→        filtered = filtered.filter((p: OrgPosition) => p.territory?.level === territoryFilter.level)
   557→      }
   558→      setPositions(filtered)
   559→    }).catch((e) => setError(e.message)).finally(() => setLoading(false))
   560→  }
   561→  useEffect(() => { loadData() }, [])
   562→
   563→  // Listen for pengurus-updated event (from sync/bulk operations)
   564→  useEffect(() => {
   565→    const handler = () => loadData()
   566→    window.addEventListener('pengurus-updated', handler)
   567→    return () => window.removeEventListener('pengurus-updated', handler)
   568→  }, [])
   569→
   570→  const handleDelete = async () => {
   571→    if (!deletePos) return
   572→    try {
   573→      await api(`/api/organization/${deletePos.id}`, { method: 'DELETE' })
   574→      addToast(`Pengurus "${deletePos.fullName}" berhasil dihapus`, 'success')
   575→      setDeletePos(null); loadData()
   576→    } catch (e: any) { addToast(e.message, 'error') }
   577→  }
   578→
   579→  if (loading) return <LoadingState />
   580→  if (error) return <ErrorState message={error} />
   581→
   582→  const canManage = user.role === 'SUPERADMIN' || user.role === 'ADMIN_DPN' ||
   583→    (user.role === 'ADMIN_DPD' && (level === 'DPD' || level === 'DPC')) ||
   584→    (user.role === 'ADMIN_DPC' && level === 'DPC')
   585→
   586→  // === Group pengurus by category ===
   587→  const sorted = [...positions].sort((a, b) => (a.order || 0) - (b.order || 0))
   588→
   589→  const getCategory = (pos: string): string => {
   590→    const p = pos.toLowerCase()
   591→    if (p.includes('ketua') && !p.includes('wakil')) return 'PIMPINAN UTAMA'
   592→    if (p.includes('wakil ketua')) return 'WAKIL KETUA'
   593→    if (p.includes('sekretaris') || p.includes('bendahara')) return 'SEKRETARIAT & KEUANGAN'
   594→    if (p.includes('bidang') || p.includes('humas') || p.includes('kaderisasi') ||
   595→        p.includes('pemuda') || p.includes('pemberdayaan') || p.includes('ekonomi') ||
   596→        p.includes('hukum') || p.includes('advokasi') || p.includes('litigasi') ||
   597→        p.includes('etik') || p.includes('kerukunan') || p.includes('lembaga') ||
   598→        p.includes('program') || p.includes('ketenagakerjaan') || p.includes('kepemudaan'))
   599→      return 'BIDANG-BIDANG'
   600→    if (p.includes('penasihat') || p.includes('pembina') || p.includes('pelindung'))
   601→      return 'DEWAN PENASIHAT'
   602→    return 'LAINNYA'
   603→  }
   604→
   605→  const categoryOrder = ['PIMPINAN UTAMA', 'WAKIL KETUA', 'SEKRETARIAT & KEUANGAN', 'BIDANG-BIDANG', 'DEWAN PENASIHAT', 'LAINNYA']
   606→  const categoryColors: Record<string, string> = {
   607→    'PIMPINAN UTAMA': 'from-red-500 to-orange-600',
   608→    'WAKIL KETUA': 'from-orange-400 to-amber-500',
   609→    'SEKRETARIAT & KEUANGAN': 'from-blue-500 to-cyan-600',
   610→    'BIDANG-BIDANG': 'from-emerald-500 to-teal-600',
   611→    'DEWAN PENASIHAT': 'from-purple-500 to-indigo-600',
   612→    'LAINNYA': 'from-slate-400 to-slate-500',
   613→  }
   614→
   615→  const grouped: Record<string, OrgPosition[]> = {}
   616→  for (const p of sorted) {
   617→    const cat = getCategory(p.positionName)
   618→    if (!grouped[cat]) grouped[cat] = []
   619→    grouped[cat].push(p)
   620→  }
   621→
   622→  // Helper: handle KTA photo upload (base64)
   623→  const handleKtaUpload = async (positionId: string, file: File) => {
   624→    if (!file.type.startsWith('image/')) { addToast('File KTA harus berupa gambar (JPG/PNG)', 'error'); return }
   625→    if (file.size > 2 * 1024 * 1024) { addToast('Ukuran KTA maksimal 2MB', 'error'); return }
   626→    try {
   627→      const buf = Buffer.from(await file.arrayBuffer())
   628→      const mime = file.type || 'image/jpeg'
   629→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   630→      await api(`/api/organization/${positionId}`, { method: 'PUT', body: JSON.stringify({ ktaPhotoUrl: dataUrl }) })
   631→      addToast('Foto KTA berhasil diupload', 'success')
   632→      loadData()
   633→    } catch (e: any) { addToast(e.message, 'error') }
   634→  }

   644→      const buf = Buffer.from(await file.arrayBuffer())
   645→      const mime = file.type || (file.name.endsWith('.pdf') ? 'application/pdf' : 'image/jpeg')
   646→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   647→      await api(`/api/organization/${positionId}`, { method: 'PUT', body: JSON.stringify({ biodataUrl: dataUrl }) })
   648→      addToast('Biodata berhasil diupload', 'success')
   649→      loadData()
   650→    } catch (e: any) { addToast(e.message, 'error') }
   651→  }

   658→        <div className="relative shrink-0">

   665→          )}
   666→          {canManage && (
   667→            <label className="absolute -bottom-1 -right-1 w-5 h-5 rounded-full bg-blue-600 hover:bg-blue-700 flex items-center justify-center cursor-pointer" title="Upload KTA">
   668→              <Upload className="w-3 h-3 text-white" />
   669→              <input type="file" accept="image/jpeg,image/png" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) handleKtaUpload(p.id, f); e.target.value = '' }} />
   670→            </label>

   680→                KTA: {p.ktaNumber}
   681→              </Badge>
   682→            </div>
   683→          )}
   684→          {p.phone && (
   685→            <a href={`https://wa.me/${p.phone.replace(/\D/g, '').replace(/^0/, '62')}`} target="_blank" rel="noopener noreferrer" className="flex items-center gap-1 mt-1 text-xs text-blue-600 hover:underline">
   686→              <Phone className="w-3 h-3" /> {p.phone}
   687→            </a>
   688→          )}
   689→          {p.email && p.email !== '-' && (
   690→            <div className="flex items-center gap-1 mt-0.5 text-xs text-muted-foreground">
   691→              <Mail className="w-3 h-3" /> {p.email}

   725→  return (

   728→        <div className="flex items-center justify-between flex-wrap gap-2">

  1030→// SECTION SK (Reusable untuk DPN/DPD/DPC)
  1031→// ============================================================
  1032→function SKSection({ level, territoryId, territoryFilter }: {
  1033→  level: string; territoryId?: string; territoryFilter?: { level: string }
  1034→}) {
  1035→  const user = useAuthStore((s) => s.user)!
  1036→  const addToast = useToastStore((s) => s.addToast)
  1037→  const [docs, setDocs] = useState<SKDocument[]>([])
  1038→  const [loading, setLoading] = useState(true)
  1039→  const [uploadOpen, setUploadOpen] = useState(false)
  1040→  const [deleteDoc, setDeleteDoc] = useState<SKDocument | null>(null)
  1041→  // OCR auto-extract state
  1042→  const [ocrLoading, setOcrLoading] = useState(false)
  1043→  const [ocrPreview, setOcrPreview] = useState<any>(null)
  1044→
  1045→  const loadData = () => {
  1046→    setLoading(true)
  1047→    api('/api/sk').then((all) => {
  1048→      let filtered = all
  1049→      if (territoryId) {
  1050→        filtered = all.filter((d: SKDocument) => d.territoryId === territoryId)
  1051→      } else if (territoryFilter) {
  1052→        filtered = all.filter((d: SKDocument) => d.territory.level === territoryFilter.level)
  1053→      }
  1054→      setDocs(filtered)
  1055→    }).catch(() => {}).finally(() => setLoading(false))
  1056→  }
  1057→  useEffect(() => { loadData() }, [])
  1058→
  1059→  const handleDelete = async () => {
  1060→    if (!deleteDoc) return
  1061→    try {
  1062→      await api(`/api/sk/${deleteDoc.id}`, { method: 'DELETE' })
  1063→      addToast('SK dihapus', 'success'); setDeleteDoc(null); loadData()
  1064→    } catch (e: any) { addToast(e.message, 'error') }
  1065→  }
  1066→
  1067→  // === Handler: Sinkronkan SK yang sudah diupload ke Struktur Pengurus ===
  1068→  // User klik tombol "Sinkronkan" per SK di daftar arsip
  1069→  // → panggil /api/sk/[id]/sync-pengurus → dapat daftar pengurus
  1070→  // → tampilkan OcrPreviewDialog → user klik tombol "Sinkronkan ke Struktur Pengurus"
  1071→  const [syncLoading, setSyncLoading] = useState<string | null>(null)
  1072→  const handleSyncPengurus = async (skId: string) => {
  1073→    setSyncLoading(skId)
  1074→    try {
  1075→      const res = await fetch(`/api/sk/${skId}/sync-pengurus`, {
  1076→        method: 'POST',
  1077→        headers: { 'x-user-id': user?.id || '' },
  1078→      })
  1079→      const data = await res.json()
  1080→      if (!res.ok || !data.success) {
  1081→        throw new Error(data.error || 'Gagal sync pengurus')
  1082→      }
  1083→
  1084→      if (data.data.pengurus && data.data.pengurus.length > 0) {
  1085→        // Tampilkan dialog preview dengan daftar pengurus
  1086→        setOcrPreview(data.data)
  1087→        addToast(`Berhasil ekstrak ${data.data.pengurus.length} pengurus dari SK. Klik tombol hijau untuk sinkronkan.`, 'success')
  1088→      } else {
  1089→        addToast('OCR selesai, tidak ada pengurus terdeteksi dari SK ini.', 'info')
  1090→      }
  1091→    } catch (e: any) {
  1092→      addToast(`Gagal sync: ${e.message}`, 'error')
  1093→    } finally {
  1094→      setSyncLoading(null)
  1095→    }
  1096→  }
  1097→
  1098→  if (loading) return <LoadingState />
  1099→
  1100→  const canManage = user.role === 'SUPERADMIN' || user.role === 'ADMIN_DPN' ||
  1101→    (user.role === 'ADMIN_DPD' && (level === 'DPD' || level === 'DPC')) ||
  1102→    (user.role === 'ADMIN_DPC' && level === 'DPC')
  1103→
  1104→  // Handler: Upload SK + OCR Extract Pengurus
  1105→  const handleOcrExtract = async (file: File) => {
  1106→    setOcrLoading(true)
  1107→    try {
  1108→      const formData = new FormData()
  1109→      formData.append('file', file)
  1110→      formData.append('territoryId', territoryId || '')
  1111→      formData.append('level', level)
  1112→
  1113→      const res = await fetch('/api/sk/extract-pengurus', {
  1114→        method: 'POST',
  1115→        headers: { 'x-user-id': user?.id || '' },
  1116→        body: formData,
  1117→      })
  1118→      const data = await res.json()
  1119→      if (!res.ok || !data.success) throw new Error(data.error || 'OCR gagal')
  1120→
  1121→      if (data.data.pengurus.length > 0) {
  1122→        setOcrPreview(data.data)
  1123→        addToast(`Berhasil ekstrak ${data.data.pengurus.length} pengurus dari SK`, 'success')
  1124→      } else {
  1125→        addToast('OCR selesai, tidak ada pengurus terdeteksi. SK tersimpan sebagai arsip.', 'info')
  1126→      }
  1127→      loadData()
  1128→    } catch (e: any) {
  1129→      addToast(e.message, 'error')
  1130→    } finally {
  1131→      setOcrLoading(false)
  1132→    }
  1133→  }
  1134→
  1135→  // Handler: Confirm extracted pengurus → bulk create records (fast, single API call)
  1136→  const handleConfirmExtract = async (pengurusList: any[]) => {
  1137→    if (!territoryId) {
  1138→      addToast('Territory ID tidak ditemukan. Pilih wilayah dulu.', 'error')
  1139→      return
  1140→    }
  1141→    try {
  1142→      // Build multiline text in format "Nama - Jabatan" for bulk API
  1143→      const pengurusText = pengurusList
  1144→        .map((p) => `${p.fullName || ''} - ${p.positionName || 'Pengurus'}`)
  1145→        .join('\n')
  1146→
  1147→      // Call bulk endpoint with territoryId + level
  1148→      const res = await fetch('/api/organization/bulk', {
  1149→        method: 'POST',
  1150→        headers: { 'Content-Type': 'application/json', 'x-user-id': user?.id || '' },
  1151→        body: JSON.stringify({
  1152→          territoryId,
  1153→          level,
  1154→          pengurusText,
  1155→        }),
  1156→      })
  1157→      const data = await res.json()
  1158→      if (!res.ok || !data.success) throw new Error(data.error)
  1159→
  1160→      const { totalCreated, totalSkipped, totalErrors } = data.data
  1161→      addToast(
  1162→        `✅ Sinkronisasi selesai! ${totalCreated} pengurus ${level} ditambahkan ke Struktur Pengurus. ${totalSkipped} duplikat di-skip. ${totalErrors} error.`,
  1163→        'success'
  1164→      )
  1165→      setOcrPreview(null)
  1166→      // Trigger reload of pengurus list di parent component
  1167→      window.dispatchEvent(new Event('pengurus-updated'))
  1168→    } catch (e: any) {
  1169→      addToast(e.message, 'error')
  1170→    }
  1171→  }
  1172→
  1173→  return (
  1174→    <Card>
  1175→      <CardHeader>
  1176→        <div className="flex items-center justify-between flex-wrap gap-2">
  1177→          <CardTitle className="flex items-center gap-2 text-base">
  1178→            <FileText className="w-4 h-4 text-indigo-600" />
  1179→            Arsip SK {level}
  1180→            <Badge variant="outline" className="text-xs">{docs.length} dokumen</Badge>
  1181→          </CardTitle>
  1182→          {canManage && (
  1183→            <div className="flex gap-2 flex-wrap">
  1184→              {/* Upload SK biasa */}
  1185→              <Button onClick={() => setUploadOpen(true)} size="sm" variant="outline">
  1186→                <Upload className="w-4 h-4 mr-1" /> Upload SK
  1187→              </Button>
  1188→              {/* Upload SK + Extract — auto-sync ke Struktur Pengurus */}
  1189→              <OcrExtractButton onFileSelected={handleOcrExtract} loading={ocrLoading} level={level} />
  1190→            </div>
  1191→          )}
  1192→        </div>
  1193→      </CardHeader>
  1194→      <CardContent>
  1195→        {docs.length === 0 ? (
  1196→          <EmptyState icon={FileText} title="Belum ada SK" description="Upload SK untuk arsip digital dengan OCR." />
  1197→        ) : (
  1198→          <div className="grid gap-2 md:grid-cols-2">
  1199→            {docs.map((d) => (
  1200→              <div key={d.id} className="group flex items-start gap-3 p-3 rounded-lg border">
  1201→                <div className="w-10 h-10 rounded-lg bg-indigo-100 flex items-center justify-center shrink-0">
  1202→                  <FileText className="w-5 h-5 text-indigo-600" />
  1203→                </div>

  1205→                  <div className="font-medium text-sm truncate">{d.title}</div>
  1206→                  <code className="text-[13px] font-mono bg-muted px-1 rounded">{d.skNumber}</code>

  1208→                    {d.issuedBy} • {formatDateID(d.issuedAt)}

  1819→  const user = useAuthStore((s) => s.user)!
  1820→  const [form, setForm] = useState({ skNumber: '', title: '', issuedAt: '', issuedBy: '' })
  1821→  const [file, setFile] = useState<File | null>(null)
  1822→  const [loading, setLoading] = useState(false)
  1823→  const [territories, setTerritories] = useState<Territory[]>([])
  1824→  const [selectedTerritory, setSelectedTerritory] = useState(territoryId || '')
  1825→
  1826→  useEffect(() => {
  1827→    if (open && !territoryId) {
  1828→      api('/api/territory').then((all) => {
  1829→        const levelMap: Record<string, string> = { DPN: 'COUNTRY', DPD: 'PROVINCE', DPC: 'REGENCY' }
  1830→        const targetLevel = territoryFilter ? levelMap[territoryFilter.level === 'COUNTRY' ? 'DPN' : territoryFilter.level === 'PROVINCE' ? 'DPD' : 'DPC'] : null
  1831→        if (targetLevel) {
  1832→          setTerritories(all.filter((t: Territory) => t.level === targetLevel && t.isActive))
  1833→        }
  1834→      }).catch(() => {})
  1835→    }
  1836→  }, [open])
  1837→
  1838→  const handleSubmit = async (e: React.FormEvent) => {
  1839→    e.preventDefault()
  1840→    const targetTerritory = territoryId || selectedTerritory
  1841→    if (!targetTerritory) { addToast('Pilih wilayah dulu', 'error'); return }
  1842→
  1843→    setLoading(true)
  1844→    try {
  1845→      const formData = new FormData()
  1846→      if (file) formData.append('file', file)
  1847→      formData.append('territoryId', targetTerritory)

  1858→      setForm({ skNumber: '', title: '', issuedAt: '', issuedBy: '' }); setFile(null)
