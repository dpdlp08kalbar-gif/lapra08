   125→  const loadData = () => {

   315→  WHATSAPP: {
   316→    label: 'WhatsApp',
   317→    charLimit: 4096,
   318→    mediaRequired: false,
   319→    features: ['Text + Image', 'Button Reply', 'List Message', 'Direct WA'],
   320→    color: 'emerald',
   321→    gradient: 'from-emerald-500 to-teal-600',
   322→  },
   323→  FACEBOOK: {
   324→    label: 'Facebook',
   325→    charLimit: 63206,
   326→    mediaRequired: false,
   327→    features: ['Text + Media', 'Link Preview', 'Reactions', 'Comments', 'Shares'],
   328→    color: 'blue',
   329→    gradient: 'from-blue-500 to-indigo-600',
   330→  },
   331→  INSTAGRAM: {
   332→    label: 'Instagram',
   333→    charLimit: 2200,
   334→    mediaRequired: true,
   335→    features: ['Image/Video Required', 'Caption', 'Hashtags', 'Stories', 'Reels'],
   336→    color: 'purple',
   337→    gradient: 'from-purple-500 to-pink-600',
   338→  },

   341→function AddBroadcastDialog({
   342→  open, onOpenChange, territories, onSuccess,
   343→}: {
   344→  open: boolean; onOpenChange: (o: boolean) => void
   345→  territories: Territory[]; onSuccess: () => void
   346→}) {
   347→  const addToast = useToastStore((s) => s.addToast)
   348→  const [form, setForm] = useState({
   349→    title: '', message: '', territoryId: '', scheduledAt: '',
   350→    imageUrl: '', videoUrl: '', linkUrl: '',
   351→  })
   352→  const [channels, setChannels] = useState<Set<string>>(new Set(['WHATSAPP']))
   353→  const [imageFile, setImageFile] = useState<File | null>(null)
   354→  const [videoFile, setVideoFile] = useState<File | null>(null)
   355→  const [loading, setLoading] = useState(false)
   356→
   357→  // Reset form on close
   358→  useEffect(() => {
   359→    if (!open) {
   360→      setForm({ title: '', message: '', territoryId: '', scheduledAt: '', imageUrl: '', videoUrl: '', linkUrl: '' })
   361→      setChannels(new Set(['WHATSAPP']))
   362→      setImageFile(null)
   363→      setVideoFile(null)
   364→    }
   365→  }, [open])
   366→
   367→  const toggleChannel = (ch: string) => {
   368→    setChannels(prev => {
   369→      const next = new Set(prev)
   370→      if (next.has(ch)) next.delete(ch)
   371→      else next.add(ch)
   372→      return next
   373→    })
   374→  }
   375→
   376→  const handleImageUpload = (file: File) => {
   377→    if (!file.type.startsWith('image/')) {
   378→      addToast('File harus gambar', 'error')
   379→      return
   380→    }
   381→    if (file.size > 5 * 1024 * 1024) {
   382→      addToast('Ukuran gambar maks 5MB', 'error')
   383→      return
   384→    }
   385→    const reader = new FileReader()
   386→    reader.onload = (e) => setForm({ ...form, imageUrl: e.target?.result as string })
   387→    reader.readAsDataURL(file)
   388→    setImageFile(file)
   389→  }
   390→
   391→  const handleVideoUpload = (file: File) => {
   392→    if (!file.type.startsWith('video/')) {
   393→      addToast('File harus video', 'error')
   394→      return
   395→    }
   396→    if (file.size > 100 * 1024 * 1024) {
   397→      addToast('Ukuran video maks 100MB', 'error')
   398→      return
   399→    }
   400→    setForm({ ...form, videoUrl: URL.createObjectURL(file) })
   401→    setVideoFile(file)
   402→  }
   403→
   404→  const handleSubmit = async (e: React.FormEvent) => {
   405→    e.preventDefault()
   406→    const channelArr = Array.from(channels)
   407→    if (channelArr.length === 0) {
   408→      addToast('Pilih minimal 1 channel', 'error')
   409→      return
   410→    }
   411→    if (channelArr.includes('INSTAGRAM') && !form.imageUrl && !form.videoUrl) {
   412→      addToast('Instagram wajib memiliki gambar atau video', 'error')
   413→      return
   414→    }
   415→    // Check char limit per channel
   416→    for (const ch of channelArr) {
   417→      const config = CHANNEL_CONFIG[ch]
   418→      if (form.message.length > config.charLimit) {
   419→        addToast(`Pesan untuk ${config.label} melebihi ${config.charLimit} karakter`, 'error')
   420→        return
   421→      }
   422→    }
   423→
   424→    setLoading(true)
   425→    try {
   426→      const formData = new FormData()
   427→      formData.append('title', form.title)
   428→      formData.append('message', form.message)
   429→      formData.append('channels', JSON.stringify(channelArr))
   430→      formData.append('targetScope', JSON.stringify(form.territoryId ? { territoryId: form.territoryId } : { all: true }))
   431→      formData.append('scheduledAt', form.scheduledAt || '')
   432→      formData.append('linkUrl', form.linkUrl || '')
   433→      if (imageFile) formData.append('image', imageFile)
   434→      if (videoFile) formData.append('video', videoFile)
   435→
   436→      const res = await fetch('/api/broadcasts', {
   437→        method: 'POST',
   438→        headers: { 'x-user-id': useAuthStore.getState().user?.id || '' },
   439→        body: formData,
   440→      })
   441→      const data = await res.json()
   442→      if (!res.ok || !data.success) throw new Error(data.error || 'Gagal')
   443→      addToast(data.message, 'success')
   444→      onSuccess()
   445→    } catch (e: any) {
   446→      addToast(e.message, 'error')
   447→    } finally {
   448→      setLoading(false)
   449→    }
   450→  }
   451→
   452→  const channelArr = Array.from(channels)
   453→
   454→  return (
   455→    <Dialog open={open} onOpenChange={onOpenChange}>
   456→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto" aria-describedby={undefined}>
   457→        <DialogHeader>
   458→          <DialogTitle className="flex items-center gap-2">
   459→            <Send className="w-5 h-5 text-orange-600" /> Multi-Channel Broadcast Composer
   460→          </DialogTitle>
   461→          <DialogDescription>Tulis sekali, kirim ke WhatsApp + Facebook + Instagram bersamaan</DialogDescription>
   462→        </DialogHeader>
   463→        <form onSubmit={handleSubmit} className="space-y-4">
   464→          {/* Channel Selector */}
   465→          <div className="space-y-3">
   466→            <Label className="text-sm font-semibold">Pilih Channel *</Label>
   467→            <div className="grid grid-cols-3 gap-3">
   468→              {Object.entries(CHANNEL_CONFIG).map(([ch, config]) => {
   469→                const isSelected = channels.has(ch)
   470→                return (
   471→                  <button
   472→                    key={ch}
   473→                    type="button"
   474→                    onClick={() => toggleChannel(ch)}
   475→                    className={`relative rounded-xl border-2 p-4 text-left transition-all ${
   476→                      isSelected ? 'shadow-md' : 'border-slate-200 hover:border-slate-300 bg-white'
   477→                    }`}
   478→                    style={isSelected ? {
   479→                      borderColor: ch === 'WHATSAPP' ? '#10b981' : ch === 'FACEBOOK' ? '#3b82f6' : '#a855f7',
   480→                      backgroundColor: ch === 'WHATSAPP' ? '#ecfdf5' : ch === 'FACEBOOK' ? '#eff6ff' : '#faf5ff',
