   361→// PROGRAM LEVEL VIEW — List dokumen per level + Upload/View/Edit/Delete
   362→// ============================================================
   363→function ProgramLevelView({
   364→  title, description, icon: Icon, accentColor, items, loading, category, level, territoryCode, territoryName, onBack, setItems,
   365→}: {
   366→  title: string
   367→  description: string
   368→  icon: any
   369→  accentColor: string
   370→  items: ProgramDoc[]
   371→  loading: boolean
   372→  category: string
   373→  level: Level
   374→  territoryCode: string
   375→  territoryName: string
   376→  onBack: () => void
   377→  setItems: React.Dispatch<React.SetStateAction<ProgramDoc[]>>
   378→}) {
   379→  const addToast = useToastStore((s) => s.addToast)
   380→  const user = useAuthStore((s) => s.user)
   381→  const isSuperAdmin = useIsSuperAdmin()
   382→  const [search, setSearch] = useState('')
   383→  const [uploadOpen, setUploadOpen] = useState(false)
   384→  const [editDoc, setEditDoc] = useState<ProgramDoc | null>(null)
   385→  const [deleteDoc, setDeleteDoc] = useState<ProgramDoc | null>(null)
   386→  const [viewingDoc, setViewingDoc] = useState<ProgramDoc | null>(null)
   387→
   388→  // === Helper: format ukuran file ===
   389→  const formatFileSize = (bytes?: number) => {
   390→    if (!bytes) return '-'
   391→    if (bytes < 1024) return `${bytes} B`
   392→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   393→    return `${(bytes / 1024 / 1024).toFixed(2)} MB`
   394→  }
   395→
   396→  // === Helper: ikon file ===
   397→  const getFileIcon = (fileType?: string | null) => {
   398→    if (!fileType) return FileText
   399→    const t = fileType.toUpperCase()
   400→    if (['JPG', 'JPEG', 'PNG', 'WEBP', 'GIF'].includes(t)) return ImageIcon
   401→    if (['MP4', 'MOV', 'WEBM'].includes(t)) return Video
   402→    if (['PDF'].includes(t)) return FileText
   403→    if (['DOC', 'DOCX', 'TXT'].includes(t)) return FileText
   404→    if (['XLS', 'XLSX', 'CSV'].includes(t)) return FileCheck
   405→    return FileText
   406→  }
   407→
   408→  // === Handler: reload list ===
   409→  const reload = () => {
   410→    Promise.all([
   411→      api('/api/program-documents').catch(() => []),
   412→      api('/api/gallery').catch(() => []),
   413→    ]).then(([allDocs, allGallery]: any[]) => {
   414→      const newDocs = (allDocs || []).filter((d: any) => d.category === category)
   415→      const legacy = (allGallery || []).filter((a: any) => a.category === category)
   416→      setItems([...newDocs, ...legacy])
   417→    })
   418→  }
   419→
   420→  // === Handler: View bukti (inline di browser) ===
   421→  const handleView = async (doc: ProgramDoc) => {
   422→    setViewingDoc(doc)
   423→    try {
   424→      // Untuk dokumen baru: pakai /api/program-documents/[id]/view
   425→      // Untuk dokumen legacy (fileUrl langsung): buka fileUrl
   426→      let blobUrl: string
   427→      if (doc.id.startsWith('progdoc_')) {
   428→        const res = await fetch(`/api/program-documents/${doc.id}/view`, {
   429→          headers: { 'x-user-id': user?.id || '' },
   430→        })
   431→        if (!res.ok) {
   432→          const errData = await res.json().catch(() => null)
   433→          throw new Error(errData?.error || `HTTP ${res.status}`)
   434→        }
   435→        const blob = await res.blob()
   436→        blobUrl = URL.createObjectURL(blob)
   437→      } else if (doc.fileUrl) {
   438→        // Legacy: fileUrl mungkin data URL atau public URL
   439→        if (doc.fileUrl.startsWith('data:')) {
   440→          const res = await fetch(doc.fileUrl)
   441→          const blob = await res.blob()
   442→          blobUrl = URL.createObjectURL(blob)
   443→        } else {
   444→          window.open(doc.fileUrl, '_blank')
   445→          return
   446→        }
   447→      } else if (doc.pdfUrl) {
   448→        // Legacy: PDF program kerja
   449→        if (doc.pdfUrl.startsWith('data:')) {
   450→          const res = await fetch(doc.pdfUrl)
   451→          const blob = await res.blob()
   452→          blobUrl = URL.createObjectURL(blob)
   453→        } else {
   454→          window.open(doc.pdfUrl, '_blank')
   455→          return
   456→        }
   457→      } else {
   458→        addToast('Dokumen tidak punya file', 'error')
   459→        return
   460→      }
   461→      window.open(blobUrl, '_blank')
   462→      setTimeout(() => URL.revokeObjectURL(blobUrl), 60000)
   463→    } catch (e: any) {
   464→      addToast(`Gagal membuka file: ${e.message}`, 'error')
   465→    } finally {
   466→      setViewingDoc(null)
   467→    }
   468→  }
   469→
   470→  // === Handler: Download file ===

   481→        const res = await fetch(doc.fileUrl)
   482→        blob = await res.blob()
   483→      } else if (doc.pdfUrl?.startsWith('data:')) {
   484→        const res = await fetch(doc.pdfUrl)
   485→        blob = await res.blob()
   486→      } else if (doc.fileUrl || doc.pdfUrl) {
   487→        // External URL
   488→        const a = window.document.createElement('a')
   489→        a.href = (doc.fileUrl || doc.pdfUrl) as string
   490→        a.download = doc.fileName || doc.title
   491→        a.target = '_blank'
   492→        window.document.body.appendChild(a)
   493→        a.click()
   494→        window.document.body.removeChild(a)
   495→        return
   496→      } else {
   497→        addToast('Tidak ada file untuk diunduh', 'error')
   498→        return
   499→      }
   500→      const blobUrl = URL.createObjectURL(blob)
   501→      const a = window.document.createElement('a')
   502→      a.href = blobUrl
   503→      a.download = doc.fileName || `${doc.title}.bin`
   504→      window.document.body.appendChild(a)
   505→      a.click()
   506→      window.document.body.removeChild(a)
   507→      setTimeout(() => URL.revokeObjectURL(blobUrl), 30000)
   508→      addToast(`File "${doc.fileName || doc.title}" didownload`, 'success')
   509→    } catch (e: any) {
   510→      addToast(`Gagal download: ${e.message}`, 'error')
   511→    }
   512→  }
   513→
   514→  if (loading) return <LoadingState message={`Memuat dokumen ${level}...`} />
   515→

   517→    !search ||
   518→    i.title?.toLowerCase().includes(search.toLowerCase()) ||
   519→    i.description?.toLowerCase().includes(search.toLowerCase()) ||

   549→          <Upload className="w-4 h-4" /> Upload Dokumen {level}
   550→        </Button>
   551→      </div>
   552→
   553→      {/* Search */}
   554→      <div className="relative max-w-md">
   555→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   556→        <Input
   557→          placeholder="Cari judul / deskripsi / lokasi..."
   558→          value={search}
   559→          onChange={(e) => setSearch(e.target.value)}
   560→          className="pl-9"
   561→        />
   562→      </div>
   563→
   564→      {/* Stats ringkas */}
   565→      <div className="grid grid-cols-4 gap-2 text-xs">
   566→        <Card className="p-2"><div className="text-muted-foreground">Total</div><div className="text-lg font-bold">{stats.total}</div></Card>
   567→        <Card className="p-2"><div className="text-muted-foreground">Selesai</div><div className="text-lg font-bold text-green-600">{stats.selesai}</div></Card>
   568→        <Card className="p-2"><div className="text-muted-foreground">Berjalan</div><div className="text-lg font-bold text-blue-600">{stats.berjalan}</div></Card>
   569→        <Card className="p-2"><div className="text-muted-foreground">Direncanakan</div><div className="text-lg font-bold text-orange-600">{stats.direncanakan}</div></Card>
   570→      </div>
   571→
   572→      {/* Empty state */}
   573→      {filtered.length === 0 ? (
   574→        <EmptyState
   575→          icon={FileText}
   576→          title={`Belum ada dokumen ${level}`}
   577→          description={`Upload dokumen bukti pelaksanaan ${level} ${territoryName}. Mendukung PDF, gambar, dokumen, video, dan audio.`}

   581→            </Button>

   935→    title: doc.title,
   936→    description: doc.description || '',

   943→  const [saving, setSaving] = useState(false)
   944→  const fileInputRef = useRef<HTMLInputElement>(null)

   954→    }

   958→      if (doc.id.startsWith('progdoc_')) {
   959→        // Dokumen baru via /api/program-documents/[id]
   960→        if (replaceFile && newFile) {
   961→          const formData = new FormData()
   962→          formData.append('file', newFile)
   963→          formData.append('title', form.title)
   964→          formData.append('description', form.description)
   965→          formData.append('location', form.location)
   966→          formData.append('date', form.date)
   967→          formData.append('status', form.status)
   968→
   969→          const res = await fetch(`/api/program-documents/${doc.id}`, {
   970→            method: 'PUT',
   971→            headers: { 'x-user-id': user?.id || '' },
   972→            body: formData,
   973→          })
   974→          const data = await res.json()
   975→          if (!res.ok || !data.success) throw new Error(data.error || `HTTP ${res.status}`)
   976→          addToast('File & metadata berhasil diperbarui', 'success')
   977→        } else {
   978→          await api(`/api/program-documents/${doc.id}`, {
   979→            method: 'PUT',
   980→            body: JSON.stringify(form),
   981→          })
   982→          addToast('Metadata berhasil diperbarui', 'success')
   983→        }
