   276→                    onEdit={setEditTerritory}

   377→function TerritoryTreeList({
   378→  territories, orgPositions, expandedIds, onToggleExpand,
   379→  onEdit, onDelete, onAddPosition, onEditPosition, onDeletePosition, canCreate,
   380→}: {
   381→  territories: Territory[]
   382→  orgPositions: OrgPosition[]
   383→  expandedIds: Set<string>
   384→  onToggleExpand: (id: string) => void
   385→  onEdit: (t: Territory) => void
   386→  onDelete: (t: Territory) => void
   387→  onAddPosition: (t: Territory) => void
   388→  onEditPosition: (p: OrgPosition) => void
   389→  onDeletePosition: (p: OrgPosition) => void
   390→  canCreate: boolean
   391→}) {
   392→  const buildTree = (items: Territory[], parentId: string | null = null): Territory[] => {
   393→    return items
   394→      .filter((t) => t.parentId === parentId)
   395→      .sort((a, b) => {
   396→        const aIdx = LEVEL_ORDER.indexOf(a.level)
   397→        const bIdx = LEVEL_ORDER.indexOf(b.level)
   398→        if (aIdx !== bIdx) return aIdx - bIdx
   399→        return a.name.localeCompare(b.name)
   400→      })
   401→  }
   402→
   403→  const roots = buildTree(territories, null)
   404→
   405→  if (roots.length === 0) {
   406→    return (
   407→      <div className="p-8 text-center text-sm text-muted-foreground">
   408→        Tidak ada data wilayah
   409→      </div>
   410→    )
   411→  }
   412→
   413→  return (
   414→    <div className="p-2 space-y-1">
   415→      {roots.map((t) => (
   416→        <TerritoryNode
   417→          key={t.id}
   418→          territory={t}
   419→          allTerritories={territories}
   420→          orgPositions={orgPositions}
   421→          depth={0}
   422→          expandedIds={expandedIds}
   423→          onToggleExpand={onToggleExpand}
   424→          onEdit={onEdit}
   425→          onDelete={onDelete}
   426→          onAddPosition={onAddPosition}
   427→          onEditPosition={onEditPosition}
   428→          onDeletePosition={onDeletePosition}
   429→          canCreate={canCreate}
   430→        />
   431→      ))}
   432→    </div>
   433→  )
   434→}

   436→function TerritoryNode({
   437→  territory, allTerritories, orgPositions, depth, expandedIds, onToggleExpand,
   438→  onEdit, onDelete, onAddPosition, onEditPosition, onDeletePosition, canCreate,
   439→}: {
   440→  territory: Territory
   441→  allTerritories: Territory[]
   442→  orgPositions: OrgPosition[]
   443→  depth: number
   444→  expandedIds: Set<string>
   445→  onToggleExpand: (id: string) => void
   446→  onEdit: (t: Territory) => void
   447→  onDelete: (t: Territory) => void
   448→  onAddPosition: (t: Territory) => void
   449→  onEditPosition: (p: OrgPosition) => void
   450→  onDeletePosition: (p: OrgPosition) => void
   451→  canCreate: boolean
   452→}) {
   453→  const children = allTerritories.filter((t) => t.parentId === territory.id)
   454→    .sort((a, b) => {
   455→      const aIdx = LEVEL_ORDER.indexOf(a.level)
   456→      const bIdx = LEVEL_ORDER.indexOf(b.level)
   457→      if (aIdx !== bIdx) return aIdx - bIdx
   458→      return a.name.localeCompare(b.name)
   459→    })
   460→  const hasChildren = children.length > 0
   461→  const isExpanded = expandedIds.has(territory.id)
   462→  const Icon = LEVEL_ICONS[territory.level] || Map
   463→  const positions = orgPositions.filter((p) => p.territoryId === territory.id && p.isActive)
   464→    .sort((a, b) => a.order - b.order)
   465→  const ketua = positions.find((p) =>
   466→    p.positionName.toLowerCase().includes('ketua') &&
   467→    !p.positionName.toLowerCase().includes('wakil')
   468→  ) || positions[0]
   469→
   470→  const canEdit = territory.canEdit !== false && canCreate
   471→
   472→  return (
   473→    <div>
   474→      <div
   475→        className={`group flex items-start gap-2 p-2 rounded-lg transition-colors ${
   476→          territory.canEdit === false ? 'bg-muted/20' : 'hover:bg-accent/50'
   477→        }`}
   478→        style={{ paddingLeft: `${depth * 20 + 8}px` }}
   479→      >
   480→        {/* Expand/Collapse button */}
   481→        <button
   482→          onClick={() => hasChildren && onToggleExpand(territory.id)}
   483→          className={`mt-0.5 w-5 h-5 flex items-center justify-center shrink-0 ${
   484→            hasChildren ? 'hover:bg-muted rounded' : 'opacity-0'
   485→          }`}
   486→        >
   487→          {hasChildren ? (
   488→            isExpanded ? <ChevronDown className="w-4 h-4" /> : <ChevronRight className="w-4 h-4" />
   489→          ) : null}
   490→        </button>
   491→
   492→        {/* Icon */}
   493→        <div className={`w-8 h-8 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${LEVEL_COLORS[territory.level]}`}>
   494→          <Icon className="w-4 h-4" />
   495→        </div>
   496→
   497→        {/* Content */}
   498→        <div className="flex-1 min-w-0">
   499→          <div className="flex items-center gap-2 flex-wrap">
   500→            <span className="font-semibold text-sm truncate">{territory.name}</span>
   501→            <Badge variant="outline" className={`text-[13px] ${LEVEL_COLORS[territory.level]}`}>
   502→              {LEVEL_LABELS[territory.level]}
   503→            </Badge>
   504→            <code className="text-[13px] font-mono bg-muted px-1.5 py-0.5 rounded text-muted-foreground">
   505→              {territory.code}
   506→            </code>
   507→            {territory.isActive ? (
   508→              <Badge variant="outline" className="text-[13px] bg-emerald-50 text-emerald-700 border-emerald-200">
   509→                Aktif
   510→              </Badge>
   511→            ) : (
   512→              <Badge variant="outline" className="text-[13px] bg-gray-50 text-gray-500 border-gray-200">
   513→                Nonaktif
   514→              </Badge>
   515→            )}
   516→            {territory.canEdit === false && canCreate && (
   517→              <Badge variant="outline" className="text-[13px] bg-amber-50 text-amber-700 border-amber-200">
   518→                Read-Only
   519→              </Badge>
   520→            )}
   521→          </div>
   522→
   523→          {/* Info: Ketua & stats */}
   524→          <div className="mt-1 flex items-center gap-3 flex-wrap text-xs text-muted-foreground">
   525→            {ketua && (
   526→              <span className="inline-flex items-center gap-1">
   527→                <Crown className="w-3 h-3 text-orange-500" />
   528→                <span className="font-medium text-foreground">{ketua.fullName}</span>
   529→                <span className="text-muted-foreground">({ketua.positionName})</span>
   530→              </span>
   531→            )}
   532→            {territory._count && (
   533→              <>
   534→                {territory._count.children > 0 && (
   535→                  <span className="inline-flex items-center gap-1">
   536→                    <Layers className="w-3 h-3" />
   537→                    {territory._count.children} sub-wilayah
   538→                  </span>
   539→                )}
   540→                <span className="inline-flex items-center gap-1">
   541→                  <Users className="w-3 h-3" />
   542→                  {territory._count.members} anggota
   543→                </span>
   544→                {territory._count.users > 0 && (
   545→                  <span className="inline-flex items-center gap-1">
   546→                    <UserCog className="w-3 h-3" />
   547→                    {territory._count.users} user
   548→                  </span>
   549→                )}
   550→              </>
   551→            )}
   552→            {positions.length > 0 && (
   553→              <span className="inline-flex items-center gap-1">
