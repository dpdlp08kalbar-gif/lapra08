   115→export function MembershipMenu() {
   116→  const user = useAuthStore((s) => s.user)!
   117→  const addToast = useToastStore((s) => s.addToast)
   118→  const [members, setMembers] = useState<Member[]>([])
   119→  const [territories, setTerritories] = useState<Territory[]>([])
   120→  const [loading, setLoading] = useState(true)
   121→  const [error, setError] = useState('')
   122→  const [search, setSearch] = useState('')
   123→  const [tab, setTab] = useState('active')
   124→  const [addOpen, setAddOpen] = useState(false)
   125→  const [detailMember, setDetailMember] = useState<Member | null>(null)
   126→  const [editMember, setEditMember] = useState<Member | null>(null)
   127→
   128→  const loadData = (status?: string) => {
   129→    setLoading(true)
   130→    const statusFilter = status || (tab === 'pending' ? 'PENDING' : tab === 'active' ? 'ACTIVE' : '')
   131→    const params = new URLSearchParams()
   132→    if (statusFilter) params.set('status', statusFilter)
   133→    if (search) params.set('search', search)
   134→
   135→    Promise.all([
   136→      api(`/api/members${params.toString() ? `?${params.toString()}` : ''}`),
   137→      api('/api/territory'),
   138→    ])
   139→      .then(([m, t]) => {
   140→        setMembers(m)
   141→        // Filter territory yang bisa DIEDIT user untuk pendaftaran anggota
   142→        // DPN: COUNTRY level (untuk input anggota DPN pusat)
   143→        // DPD: PROVINCE + REGENCY level di provinsinya
   144→        // DPC: REGENCY level sendiri
   145→        setTerritories(t.filter((tt: Territory) => tt.canEdit && tt.isActive))
   146→      })
   147→      .catch((e) => setError(e.message))
   148→      .finally(() => setLoading(false))
   149→  }
   150→
   151→  useEffect(() => {
   152→    loadData()
   153→  }, [tab, search])
   154→
   155→  const handleAction = async (member: Member, action: string) => {
   156→    try {
   157→      await api(`/api/members/${member.id}`, {
   158→        method: 'PATCH',
   159→        body: JSON.stringify({ action }),
   160→      })
   161→      addToast(`Anggota berhasil ${action === 'verify' ? 'diverifikasi' : action === 'activate' ? 'diaktifkan' : action === 'reject' ? 'ditolak' : 'dinonaktifkan'}`, 'success')
   162→      loadData()
   163→    } catch (e: any) {
   164→      addToast(e.message, 'error')
   165→    }
   166→  }
   167→
   168→  const handleDelete = async (member: Member) => {
   169→    if (!confirm(`Hapus anggota ${member.fullName}? Tindakan ini tidak dapat dibatalkan.`)) return
   170→    try {
   171→      await api(`/api/members/${member.id}`, { method: 'DELETE' })
   172→      addToast('Anggota berhasil dihapus', 'success')
   173→      loadData()
   174→    } catch (e: any) {
   175→      addToast(e.message, 'error')
   176→    }
   177→  }
   178→
   179→  if (loading) return <LoadingState />
   180→  if (error) return <ErrorState message={error} />
   181→
   182→  const counts = {
   183→    pending: members.filter((m) => m.status === 'PENDING').length,
   184→    active: members.filter((m) => m.status === 'ACTIVE').length,
   185→    all: members.length,
   186→  }
   187→
   188→  return (
   189→    <div className="space-y-6">
   190→      <PageHeader
   191→        title="Data Keanggotaan"
   192→        description="Manajemen anggota lengkap dengan generator nomor KTA otomatis"
   193→        icon={Users}
   194→        actions={
   195→          <Button onClick={() => setAddOpen(true)} className="bg-gradient-to-r from-orange-600 to-red-600 text-white">
   196→            <UserPlus className="w-4 h-4 mr-2" />
   197→            Input Anggota
   198→          </Button>
   199→        }
   200→      />
   201→
   202→      {/* Search bar */}
   203→      <div className="flex items-center gap-2">
   204→        <div className="relative flex-1 max-w-md">
   205→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   206→          <Input
   207→            placeholder="Cari nama, NIK, nomor KTA, atau WhatsApp..."
   208→            value={search}
   209→            onChange={(e) => setSearch(e.target.value)}
   210→            className="pl-10"
   211→          />
   212→        </div>
   213→      </div>
   214→
   215→      <Tabs value={tab} onValueChange={setTab}>
   216→        <TabsList>
   217→          <TabsTrigger value="active">
   218→            <CheckCircle2 className="w-4 h-4 mr-2" />
   219→            Anggota Aktif ({counts.active})
   220→          </TabsTrigger>
   221→          <TabsTrigger value="pending">
   222→            <Clock className="w-4 h-4 mr-2" />
   223→            Antrean Verifikasi ({counts.pending})
   224→          </TabsTrigger>
   225→          <TabsTrigger value="all">
   226→            <Users className="w-4 h-4 mr-2" />
   227→            Semua ({counts.all})
   228→          </TabsTrigger>
   229→        </TabsList>
   230→
   231→        <TabsContent value={tab} className="mt-4">
   232→          {members.length === 0 ? (
   233→            <EmptyState
   234→              icon={Users}
   235→              title="Belum ada data anggota"
   236→              description="Klik 'Input Anggota' untuk mulai menambahkan data keanggotaan."
   237→              action={
   238→                <Button onClick={() => setAddOpen(true)} className="bg-gradient-to-r from-orange-600 to-red-600 text-white">
   239→                  <UserPlus className="w-4 h-4 mr-2" />
   240→                  Input Anggota
   241→                </Button>
   242→              }
   243→            />
   244→          ) : (
   245→            <Card>
   246→              <CardContent className="p-0">
   247→                <div className="overflow-x-auto">
   248→                  <Table>
   249→                    <TableHeader>
   250→                      <TableRow>
   251→                        <TableHead>Nomor KTA</TableHead>
   252→                        <TableHead>Nama</TableHead>
   253→                        <TableHead>WhatsApp</TableHead>
   254→                        <TableHead>Wilayah</TableHead>
   255→                        <TableHead>Status</TableHead>
   256→                        <TableHead>Terdaftar</TableHead>
   257→                        <TableHead className="text-right">Aksi</TableHead>
   258→                      </TableRow>
   259→                    </TableHeader>
   260→                    <TableBody>
   261→                      {members.map((m) => {
   262→                        const status = STATUS_CONFIG[m.status]
   263→                        const StatusIcon = status.icon
   264→                        const canEdit = m.canEdit !== false // default true jika undefined
   265→                        // Tentukan level berdasarkan territory
   266→                        const levelLabel =
   267→                          m.territory?.level === 'COUNTRY' ? 'DPN' :
   268→                          
   269→                          m.territory?.level === 'PROVINCE' ? 'DPD' :
   270→                          'DPC'
   271→                        return (
   272→                          <TableRow key={m.id} className={!canEdit ? 'bg-muted/30' : ''}>
   273→                            <TableCell>
   274→                              <code className="text-xs font-mono bg-muted px-1.5 py-0.5 rounded">
   275→                                {m.memberNumber}
   276→                              </code>
   277→                              <div className="text-[13px] text-muted-foreground mt-0.5">{levelLabel}</div>
   278→                            </TableCell>
   279→                            <TableCell className="font-medium">
   280→                              <div>
   281→                                {m.fullName}
   282→                                {m.nik && (
   283→                                  <div className="text-xs text-muted-foreground">
   284→                                    NIK: {m.nik}
   285→                                  </div>
   286→                                )}
   287→                              </div>
   288→                            </TableCell>
   289→                            <TableCell>{m.phone}</TableCell>
   290→                            <TableCell>
   291→                              <Badge variant="outline" className="text-xs">
   292→                                {m.territory.name}
   293→                              </Badge>
   294→                              {!canEdit && (
   295→                                <Badge variant="outline" className="text-[13px] ml-1 bg-amber-50 text-amber-700 border-amber-200">
   296→                                  Read-Only
   297→                                </Badge>
   298→                              )}
   299→                            </TableCell>
   300→                            <TableCell>
   301→                              <Badge variant="outline" className={`text-xs ${status.color}`}>
   302→                                <StatusIcon className="w-3 h-3 mr-1" />
   303→                                {status.label}
   304→                              </Badge>
   305→                            </TableCell>
   306→                            <TableCell className="text-xs text-muted-foreground">
   307→                              {formatDateID(m.registeredAt)}
   308→                            </TableCell>
   309→                            <TableCell className="text-right">
   310→                              <DropdownMenu>
   311→                                <DropdownMenuTrigger asChild>
   312→                                  <Button variant="ghost" size="sm" className="h-8 w-8 p-0">
   313→                                    <MoreVertical className="w-4 h-4" />
   314→                                  </Button>
   315→                                </DropdownMenuTrigger>
   316→                                <DropdownMenuContent align="end">
   317→                                  <DropdownMenuItem onClick={() => setDetailMember(m)}>
   318→                                    <Eye className="w-4 h-4 mr-2" /> Lihat Detail
   319→                                  </DropdownMenuItem>
   320→                                  {canEdit ? (
   321→                                    <>
   322→                                      <DropdownMenuItem onClick={() => setEditMember(m)}>
   323→                                        <Edit className="w-4 h-4 mr-2" /> Edit Data
   324→                                      </DropdownMenuItem>
   325→                                      {m.status === 'PENDING' && (
   326→                                        <>
   327→                                          <DropdownMenuSeparator />
   328→                                          <DropdownMenuItem onClick={() => handleAction(m, 'verify')}>
   329→                                            <ShieldCheck className="w-4 h-4 mr-2 text-blue-600" /> Verifikasi
   330→                                          </DropdownMenuItem>
   331→                                          <DropdownMenuItem onClick={() => handleAction(m, 'activate')}>
   332→                                            <CheckCircle2 className="w-4 h-4 mr-2 text-emerald-600" /> Aktifkan
   333→                                          </DropdownMenuItem>
   334→                                          <DropdownMenuItem onClick={() => handleAction(m, 'reject')}>
   335→                                            <XCircle className="w-4 h-4 mr-2 text-red-600" /> Tolak
   336→                                          </DropdownMenuItem>
   337→                                        </>
   338→                                      )}
   339→                                      <DropdownMenuSeparator />
   340→                                      <DropdownMenuItem
   341→                                        onClick={() => handleDelete(m)}
   342→                                        className="text-red-600"
   343→                                      >
   344→                                        <Trash2 className="w-4 h-4 mr-2" /> Hapus
   345→                                      </DropdownMenuItem>
   346→                                    </>
   347→                                  ) : (
   348→                                    <DropdownMenuItem disabled className="text-muted-foreground">
   349→                                      <Lock className="w-4 h-4 mr-2" /> Read-Only (Lihat saja)
   350→                                    </DropdownMenuItem>
   351→                                  )}
   352→                                  <DropdownMenuItem onClick={() => window.print()}>
   353→                                    <CreditCard className="w-4 h-4 mr-2" /> Cetak KTA
   354→                                  </DropdownMenuItem>
   355→                                </DropdownMenuContent>
   356→                              </DropdownMenu>
   357→                            </TableCell>
   358→                          </TableRow>
   359→                        )
   360→                      })}
   361→                    </TableBody>
   362→                  </Table>
   363→                </div>
   364→              </CardContent>
   365→            </Card>
   366→          )}
   367→        </TabsContent>
   368→      </Tabs>
   369→
   370→      {/* Add Member Dialog */}
   371→      <AddMemberDialog
   372→        open={addOpen}
   373→        onOpenChange={setAddOpen}
   374→        territories={territories}
   375→        onSuccess={() => {
   376→          loadData()
   377→          setAddOpen(false)
   378→        }}
   379→      />
   380→
   381→      {/* Detail Member Dialog */}
   382→      <DetailMemberDialog member={detailMember} onClose={() => setDetailMember(null)} />
   383→
   384→      {/* Edit Member Dialog */}
   385→      <EditMemberDialog
   386→        member={editMember}
   387→        territories={territories}
   388→        onClose={() => setEditMember(null)}
   389→        onSuccess={() => {
   390→          loadData()
   391→          setEditMember(null)
   392→        }}
   393→      />
   394→    </div>
   395→  )
   396→}
   397→
   398→function AddMemberDialog({
   399→  open,
   400→  onOpenChange,
   401→  territories,
   402→  onSuccess,
   403→}: {
   404→  open: boolean
   405→  onOpenChange: (open: boolean) => void
   406→  territories: Territory[]
   407→  onSuccess: () => void
   408→}) {
   409→  const addToast = useToastStore((s) => s.addToast)
   410→  const [form, setForm] = useState({
   411→    fullName: '',
   412→    nik: '',
   413→    passportNumber: '',
   414→    phone: '',
   415→    email: '',
   416→    address: '',
   417→    shirtSize: '',
   418→    profession: '',
   419→    gender: '',
   420→    birthDate: '',
   421→    birthPlace: '',
   422→    bloodType: '',
   423→    maritalStatus: '',
   424→    territoryId: '',
   425→    isInternational: false,
   426→  })
   427→  const [loading, setLoading] = useState(false)
   428→
   429→  const handleSubmit = async (e: React.FormEvent) => {
   430→    e.preventDefault()
   431→    setLoading(true)
   432→    try {
   433→      const payload = form.isInternational
   434→        ? {
   435→            fullName: form.fullName,
   436→            passportNumber: form.passportNumber,
   437→            phone: form.phone,
   438→            email: form.email,
   439→            address: form.address,
   440→            territoryId: form.territoryId,
   441→            status: 'PENDING',
   442→          }
   443→        : {
   444→            fullName: form.fullName,
   445→            nik: form.nik,
   446→            phone: form.phone,
   447→            email: form.email,
   448→            address: form.address,
   449→            shirtSize: form.shirtSize,
   450→            profession: form.profession,
   451→            gender: form.gender,
   452→            birthDate: form.birthDate || undefined,
   453→            birthPlace: form.birthPlace,
   454→            bloodType: form.bloodType,
   455→            maritalStatus: form.maritalStatus,
   456→            territoryId: form.territoryId,
   457→            status: 'PENDING',
   458→          }
   459→
   460→      await api('/api/members', {
   461→        method: 'POST',
   462→        body: JSON.stringify(payload),
   463→      })
   464→      addToast('Anggota baru berhasil ditambahkan. Nomor KTA otomatis dihasilkan.', 'success')
   465→      setForm({
   466→        fullName: '',
   467→        nik: '',
   468→        passportNumber: '',
   469→        phone: '',
   470→        email: '',
   471→        address: '',
   472→        shirtSize: '',
   473→        profession: '',
   474→        gender: '',
   475→        birthDate: '',
   476→        birthPlace: '',
   477→        bloodType: '',
   478→        maritalStatus: '',
   479→        territoryId: '',
   480→        isInternational: false,
   481→      })
   482→      onSuccess()
   483→    } catch (err: any) {
   484→      addToast(err.message, 'error')
   485→    } finally {
   486→      setLoading(false)
   487→    }
   488→  }
   489→
   490→  return (
   491→    <Dialog open={open} onOpenChange={onOpenChange}>
   492→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   493→        <DialogHeader>
   494→          <DialogTitle>Input Anggota Baru</DialogTitle>
   495→          <DialogDescription>
   496→            Nomor KTA akan dihasilkan otomatis dengan format:{' '}
   497→            <code className="text-xs bg-muted px-1.5 py-0.5 rounded font-mono">
   498→              LAPRA08.[NEGARA].[PROVINSI].[KAB/KOTA].[TAHUN].[URUT]
   499→            </code>
   500→          </DialogDescription>
   501→        </DialogHeader>
   502→        <form onSubmit={handleSubmit} className="space-y-4">
   503→          {/* Toggle domestik/internasional */}
   504→          <div className="flex gap-2 p-1 bg-muted rounded-lg">
   505→            <button
   506→              type="button"
   507→              onClick={() => setForm({ ...form, isInternational: false })}
   508→              className={`flex-1 py-2 px-3 rounded-md text-sm font-medium transition-colors ${
   509→                !form.isInternational ? 'bg-background shadow-sm' : 'text-muted-foreground'
   510→              }`}
   511→            >
   512→              Skema Domestik (Kalbar)
   513→            </button>
   514→            <button
   515→              type="button"
   516→              onClick={() => setForm({ ...form, isInternational: true })}
   517→              className={`flex-1 py-2 px-3 rounded-md text-sm font-medium transition-colors ${
   518→                form.isInternational ? 'bg-background shadow-sm' : 'text-muted-foreground'
   519→              }`}
   520→            >
   521→              Skema Internasional
   522→            </button>
   523→          </div>
   524→
   525→          <div className="space-y-2">
   526→            <Label>Wilayah Kepengurusan *</Label>
   527→            <Select
   528→              value={form.territoryId}
   529→              onValueChange={(v) => setForm({ ...form, territoryId: v })}
   530→              required
   531→            >
   532→              <SelectTrigger>
   533→                <SelectValue placeholder="Pilih wilayah..." />
   534→              </SelectTrigger>
   535→              <SelectContent>
   536→                {territories.map((t) => {
   537→                  const levelLabel =
   538→                    t.level === 'COUNTRY' ? 'DPN' :
   539→                    
   540→                    t.level === 'PROVINCE' ? 'DPD' :
   541→                    'DPC'
   542→                  return (
   543→                    <SelectItem key={t.id} value={t.id}>
   544→                      [{levelLabel}] {t.name} ({t.code})
   545→                    </SelectItem>
   546→                  )
   547→                })}
   548→              </SelectContent>
   549→            </Select>
   550→            <p className="text-xs text-muted-foreground">
   551→              Pilih wilayah sesuai tingkat kepengurusan Anda. Nomor KTA otomatis sesuai format masing-masing level.
   552→            </p>
   553→          </div>
   554→
   555→          {/* Info KTA format preview */}
   556→          {form.territoryId && (
   557→            <div className="rounded-lg bg-blue-50 border border-blue-200 p-3 text-xs text-blue-800">
   558→              <strong>Preview Nomor KTA:</strong> Akan di-generate otomatis dengan format:
   559→              <ul className="mt-1 ml-4 list-disc">
   560→                <li>DPN (Pusat): LAPRA08.ID.00.00.26.0000X</li>
   561→                <li>DPD (Provinsi): LAPRA08.ID.61.00.26.0000X</li>
   562→                <li>DPC (Kab/Kota): LAPRA08.ID.61.71.26.0000X</li>
   563→                <li>Internasional: LAPRA08.US.00.LAX.26.0000X</li>
   564→              </ul>
   565→            </div>
   566→          )}
   567→
   568→          <div className="grid grid-cols-2 gap-3">
   569→            <div className="space-y-2 col-span-2">
   570→              <Label>Nama Lengkap *</Label>
   571→              <Input
   572→                value={form.fullName}
   573→                onChange={(e) => setForm({ ...form, fullName: e.target.value })}
   574→                placeholder={form.isInternational ? 'Full Name (as in Passport)' : 'Nama sesuai KTP'}
   575→                required
   576→              />

   840→function EditMemberDialog({
   841→  member,
   842→  territories,
   843→  onClose,
   844→  onSuccess,
   845→}: {
   846→  member: Member | null
   847→  territories: Territory[]
   848→  onClose: () => void
   849→  onSuccess: () => void
   850→}) {
   851→  const addToast = useToastStore((s) => s.addToast)
   852→  const [form, setForm] = useState<any>({})
   853→  const [loading, setLoading] = useState(false)
   854→
   855→  useEffect(() => {
   856→    if (member) {
   857→      setForm({
   858→        fullName: member.fullName,
   859→        nik: member.nik || '',
   860→        passportNumber: member.passportNumber || '',
   861→        phone: member.phone,
   862→        email: member.email || '',
   863→        address: member.address || '',
   864→        shirtSize: member.shirtSize || '',
   865→        profession: member.profession || '',
   866→        gender: member.gender || '',
   867→        birthDate: member.birthDate ? member.birthDate.split('T')[0] : '',
   868→        birthPlace: member.birthPlace || '',
   869→        bloodType: member.bloodType || '',
   870→        maritalStatus: member.maritalStatus || '',
   871→        territoryId: member.territoryId,
   872→        status: member.status,
   873→      })
   874→    }
   875→  }, [member])
   876→
   877→  if (!member) return null
   878→

   880→    e.preventDefault()
   881→    setLoading(true)

   883→      await api(`/api/members/${member.id}`, {
   884→        method: 'PUT',
   885→        body: JSON.stringify(form),

   899→        <DialogHeader>
   900→          <DialogTitle>Edit Data Anggota</DialogTitle>
   901→          <DialogDescription>
   902→            Edit data <strong>{member.fullName}</strong> ({member.memberNumber})
   903→          </DialogDescription>
   904→        </DialogHeader>
   905→        <form onSubmit={handleSubmit} className="space-y-4">
   906→          <div className="grid grid-cols-2 gap-3">
   907→            <div className="space-y-2 col-span-2">
   908→              <Label>Nama Lengkap</Label>

   910→                value={form.fullName || ''}

   913→              />

   917→                <Label>NIK</Label>

   924→            <div className="space-y-2">
