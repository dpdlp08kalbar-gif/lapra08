    14→// TERRITORY - Hierarki Wilayah 3 Level (Sesuai Struktur LAPRA 08)
    15→// Level: COUNTRY | PROVINCE | REGENCY | DISTRICT | VILLAGE
    16→//   COUNTRY  = Negara (Indonesia, USA, dll) - DPN pusat
    17→//   PROVINCE = Provinsi / Negara LN - DPD
    18→//   REGENCY  = Kabupaten/Kota - DPC
    19→// ============================================================
    20→model Territory {
    21→  id          String      @id @default(cuid())
    22→  code        String      @unique // Kode wilayah: ID, 61, 6171, US, LAX, dst
    23→  name        String
    24→  level       String      // COUNTRY | PROVINCE | REGENCY | DISTRICT | VILLAGE
    25→  category    String      @default("DOMESTIC") // DOMESTIC | INTERNATIONAL
    26→  parentId    String?
    27→  parent      Territory?  @relation("TerritoryHierarchy", fields: [parentId], references: [id])
    28→  children    Territory[] @relation("TerritoryHierarchy")
    29→  isActive    Boolean     @default(true)
    30→  metadata    String?     // JSON string untuk data tambahan (koordinat, populasi, dll)
    31→  createdAt   DateTime    @default(now())
    32→  updatedAt   DateTime    @updatedAt
    33→
    34→  users       User[]
    35→  members     Member[]
    36→  assets      Asset[]
    37→  events      Event[]
    38→  announcements Announcement[]
    39→  financeTxns FinanceTransaction[]
    40→  distributionsFrom Distribution[] @relation("DistFrom")
    41→  distributionsTo   Distribution[] @relation("DistTo")
    42→  orgPositions OrgPosition[]
    43→  skDocuments SKDocument[]
    44→  ktaApplications KtaApplication[]
    45→  polls         Poll[]
    46→  crisisZones   CrisisZone[]
    47→  voterContacts VoterContact[]

    49→  essayPolls    EssayPoll[]
    50→}
