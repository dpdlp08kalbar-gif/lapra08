    98→function AgendaTab() {
    99→  const addToast = useToastStore((s) => s.addToast)
   100→  const [events, setEvents] = useState<EventItem[]>([])
   101→  const [territories, setTerritories] = useState<Territory[]>([])
   102→  const [loading, setLoading] = useState(true)
   103→  const [error, setError] = useState('')
   104→  const [addOpen, setAddOpen] = useState(false)
   105→
   106→  const loadData = () => {
   107→    setLoading(true)
   108→    Promise.all([api('/api/events'), api('/api/territory')])
   109→      .then(([e, t]) => { setEvents(e); setTerritories(t) })
   110→      .catch((err) => setError(err.message))
   111→      .finally(() => setLoading(false))
   112→  }
   113→
   114→  useEffect(() => { loadData() }, [])
   115→
   116→  if (loading) return <LoadingState />
   117→  if (error) return <ErrorState message={error} />
   118→
   119→  const upcoming = events.filter((e) => e.status === 'SCHEDULED')
   120→  const completed = events.filter((e) => e.status === 'COMPLETED')
   121→
   122→  return (
   123→    <div className="space-y-4">
   124→      <div className="grid gap-4 md:grid-cols-3">
   125→        <StatCard label="Total Event" value={events.length} icon={CalendarDays} color="blue" />
   126→        <StatCard label="Mendatang" value={upcoming.length} icon={Activity} color="orange" />
   127→        <StatCard label="Selesai" value={completed.length} icon={CheckSquare} color="emerald" />
   128→      </div>
   129→
   130→      <div className="flex justify-end">
   131→        <Button onClick={() => setAddOpen(true)} className="bg-gradient-to-r from-orange-600 to-red-600 text-white">
   132→          <Plus className="w-4 h-4 mr-2" /> Buat Event Baru
   133→        </Button>
   134→      </div>
   135→
   136→      {events.length === 0 ? (
   137→        <EmptyState icon={CalendarDays} title="Belum ada event" description="Buat agenda kegiatan organisasi." />
   138→      ) : (
   139→        <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-3">
   140→          {events.map((e) => (
   141→            <Card key={e.id} className="hover:shadow-md transition-shadow">
   142→              <CardContent className="p-4">
   143→                <div className="flex items-start justify-between gap-2 mb-2">
   144→                  <Badge variant="outline" className="text-xs bg-orange-50 text-orange-700 border-orange-200">
   145→                    {TYPE_LABELS[e.type] || e.type}
   146→                  </Badge>
   147→                  <Badge variant="outline" className={`text-xs ${
   148→                    e.status === 'COMPLETED' ? 'bg-emerald-50 text-emerald-700 border-emerald-200' :
   149→                    e.status === 'ONGOING' ? 'bg-blue-50 text-blue-700 border-blue-200' :
   150→                    e.status === 'CANCELLED' ? 'bg-red-50 text-red-700 border-red-200' :
   151→                    'bg-amber-50 text-amber-700 border-amber-200'
   152→                  }`}>
   153→                    {STATUS_LABELS[e.status] || e.status}
   154→                  </Badge>
   155→                </div>
   156→                <div className="font-bold text-sm mb-2 line-clamp-2">{e.title}</div>
   157→                {e.description && (
   158→                  <p className="text-xs text-muted-foreground mb-3 line-clamp-2">{e.description}</p>
   159→                )}
   160→                <div className="space-y-1 text-xs text-muted-foreground">
   161→                  <div className="flex items-center gap-1.5">
   162→                    <Calendar className="w-3 h-3" />
   163→                    {formatDateTimeID(e.startDate)}
   164→                  </div>
   165→                  {e.location && (
   166→                    <div className="flex items-center gap-1.5">
   167→                      <MapPin className="w-3 h-3" />
   168→                      {e.location}
   169→                    </div>
   170→                  )}
   171→                  <div className="flex items-center gap-1.5">
   172→                    <Users className="w-3 h-3" />
   173→                    {e._count.attendance} hadir / {e.targetAttendance || '?'} target
   174→                  </div>
   175→                  <div className="flex items-center gap-1.5">
   176→                    <FileText className="w-3 h-3" />
   177→                    {e._count.reports} laporan
   178→                  </div>
   179→                </div>
   180→              </CardContent>
   181→            </Card>
   182→          ))}
   183→        </div>
   184→      )}
   185→
   186→      <AddEventDialog
   187→        open={addOpen}
   188→        onOpenChange={setAddOpen}
   189→        territories={territories}
   190→        onSuccess={() => { loadData(); setAddOpen(false) }}
   191→      />
   192→    </div>
   193→  )
   194→}

   196→function AddEventDialog({
   197→  open, onOpenChange, territories, onSuccess,
   198→}: {
   199→  open: boolean; onOpenChange: (o: boolean) => void
   200→  territories: Territory[]; onSuccess: () => void
   201→}) {
   202→  const addToast = useToastStore((s) => s.addToast)
   203→  const [form, setForm] = useState({
   204→    title: '', description: '', type: 'PELANTIKAN',
   205→    startDate: '', endDate: '', location: '', territoryId: '', targetAttendance: 0,
   206→  })
   207→  const [loading, setLoading] = useState(false)
   208→
   209→  const handleSubmit = async (e: React.FormEvent) => {
   210→    e.preventDefault()
   211→    setLoading(true)
   212→    try {
   213→      await api('/api/events', {
   214→        method: 'POST',
   215→        body: JSON.stringify({
   216→          ...form,
   217→          startDate: new Date(form.startDate).toISOString(),
   218→          endDate: form.endDate ? new Date(form.endDate).toISOString() : undefined,
   219→          targetAttendance: parseInt(form.targetAttendance.toString()) || undefined,
   220→        }),
   221→      })
   222→      addToast('Event baru berhasil dibuat', 'success')
   223→      setForm({ title: '', description: '', type: 'PELANTIKAN', startDate: '', endDate: '', location: '', territoryId: '', targetAttendance: 0 })
   224→      onSuccess()
   225→    } catch (e: any) {
   226→      addToast(e.message, 'error')
   227→    } finally {
   228→      setLoading(false)
   229→    }
   230→  }
   231→
   232→  return (
   233→    <Dialog open={open} onOpenChange={onOpenChange}>
   234→      <DialogContent className="max-w-lg">
   235→        <DialogHeader>
   236→          <DialogTitle>Buat Event Baru</DialogTitle>
   237→        </DialogHeader>
   238→        <form onSubmit={handleSubmit} className="space-y-3">
   239→          <div className="space-y-2">
   240→            <Label>Judul Event *</Label>
   241→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} required />
   242→          </div>
   243→          <div className="grid grid-cols-2 gap-3">
   244→            <div className="space-y-2">
   245→              <Label>Tipe *</Label>
   246→              <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   247→                <SelectTrigger><SelectValue /></SelectTrigger>
   248→                <SelectContent>
   249→                  {Object.entries(TYPE_LABELS).map(([k, v]) => (
   250→                    <SelectItem key={k} value={k}>{v}</SelectItem>
   251→                  ))}
   252→                </SelectContent>
   253→              </Select>
   254→            </div>
   255→            <div className="space-y-2">
   256→              <Label>Target Hadir</Label>
   257→              <Input type="number" value={form.targetAttendance} onChange={(e) => setForm({ ...form, targetAttendance: parseInt(e.target.value) || 0 })} />
   258→            </div>
   259→          </div>
   260→          <div className="grid grid-cols-2 gap-3">
   261→            <div className="space-y-2">
   262→              <Label>Tanggal Mulai *</Label>
   263→              <Input type="datetime-local" value={form.startDate} onChange={(e) => setForm({ ...form, startDate: e.target.value })} required />
   264→            </div>
   265→            <div className="space-y-2">
   266→              <Label>Tanggal Selesai</Label>
   267→              <Input type="datetime-local" value={form.endDate} onChange={(e) => setForm({ ...form, endDate: e.target.value })} />
   268→            </div>
   269→          </div>
   270→          <div className="space-y-2">
   271→            <Label>Lokasi</Label>
   272→            <Input value={form.location} onChange={(e) => setForm({ ...form, location: e.target.value })} placeholder="cth: Aula Kantor Walikota" />
   273→          </div>
   274→          <div className="space-y-2">
   275→            <Label>Wilayah *</Label>
   276→            <Select value={form.territoryId} onValueChange={(v) => setForm({ ...form, territoryId: v })}>
   277→              <SelectTrigger><SelectValue placeholder="Pilih wilayah..." /></SelectTrigger>
   278→              <SelectContent>
   279→                {territories.map((t) => (
   280→                  <SelectItem key={t.id} value={t.id}>{t.name} ({t.code})</SelectItem>
   281→                ))}
   282→              </SelectContent>
   283→            </Select>
   284→          </div>
   285→          <div className="space-y-2">
   286→            <Label>Deskripsi</Label>
   287→            <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} />
   288→          </div>
   289→          <DialogFooter>
   290→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>Batal</Button>
   291→            <Button type="submit" disabled={loading}>{loading ? 'Menyimpan...' : 'Simpan Event'}</Button>
   292→          </DialogFooter>
   293→        </form>
   294→      </DialogContent>
   295→    </Dialog>
   296→  )
   297→}

   299→function AttendanceTab() {
   300→  const addToast = useToastStore((s) => s.addToast)
   301→  const user = useAuthStore((s: any) => s.user)
   302→  const [mode, setMode] = useState<'scheduled' | 'quick' | 'manual' | 'csv'>('scheduled')
   303→  const [events, setEvents] = useState<EventItem[]>([])
   304→  const [members, setMembers] = useState<any[]>([])
   305→  const [selectedEvent, setSelectedEvent] = useState<string>('')
   306→  const [attendance, setAttendance] = useState<any[]>([])
   307→  const [loading, setLoading] = useState(true)
   308→  const [submitting, setSubmitting] = useState(false)
   309→
   310→  // === Quick Event form state ===
   311→  const [quickForm, setQuickForm] = useState({
   312→    title: '',
   313→    date: new Date().toISOString().slice(0, 16),
   314→    location: '',
   315→    type: 'LAINNYA',
   316→    notes: '',
   317→  })
   318→
   319→  // === Manual names textarea ===
   320→  const [namesText, setNamesText] = useState('')
   321→
   322→  // === CSV upload state ===
   323→  const [csvFile, setCsvFile] = useState<File | null>(null)
   324→  const [csvPreview, setCsvPreview] = useState<any[]>([])
   325→
   326→  useEffect(() => {
   327→    api('/api/events').then((e) => {
   328→      setEvents(e)
   329→      setLoading(false)
   330→    }).catch(() => setLoading(false))
   331→  }, [])
   332→
   333→  useEffect(() => {
   334→    if (selectedEvent) {
   335→      api(`/api/members?status=ACTIVE`).then(setMembers).catch(() => {})
   336→      api(`/api/attendance?eventId=${selectedEvent}`).then(setAttendance).catch(() => {})
   337→    }
   338→  }, [selectedEvent])
   339→
   340→  const handleMarkAttendance = async (memberId: string, status: string) => {
   341→    if (!selectedEvent) return
   342→    try {
   343→      await api('/api/attendance', {
   344→        method: 'POST',
   345→        body: JSON.stringify({ eventId: selectedEvent, memberIds: [memberId], status }),
   346→      })
   347→      api(`/api/attendance?eventId=${selectedEvent}`).then(setAttendance)
   348→      addToast(`Kehadiran dicatat: ${status}`, 'success')
   349→    } catch (e: any) {
   350→      addToast(e.message, 'error')
   351→    }
   352→  }
   353→
   354→  // === Submit Quick Event + Attendees (manual names) ===
   355→  const handleQuickEventSubmit = async () => {
   356→    if (!quickForm.title || !quickForm.date) {
   357→      addToast('Nama acara dan tanggal wajib diisi', 'error')
   358→      return
   359→    }
   360→    const names = namesText.split('\n').map((n) => n.trim()).filter((n) => n.length >= 2)
   361→    if (names.length === 0) {
   362→      addToast('Daftar peserta wajib diisi (minimal 1 nama, satu per baris)', 'error')
   363→      return
   364→    }
   365→
   366→    setSubmitting(true)
   367→    try {
   368→      const res = await fetch('/api/attendance/bulk', {
   369→        method: 'POST',
   370→        headers: { 'Content-Type': 'application/json', 'x-user-id': user?.id || '' },
   371→        body: JSON.stringify({
   372→          mode: 'quick_event',
   373→          event: {
   374→            title: quickForm.title,
   375→            date: quickForm.date,
   376→            location: quickForm.location,
   377→            type: quickForm.type,
   378→            notes: quickForm.notes,
   379→          },
   380→          attendees: names.map((n) => ({ name: n, status: 'PRESENT' })),
   381→        }),
   382→      })
   383→      const data = await res.json()
   384→      if (!res.ok || !data.success) throw new Error(data.error)
   385→
   386→      addToast(data.message, 'success')
   387→      // Reset form
   388→      setQuickForm({ title: '', date: new Date().toISOString().slice(0, 16), location: '', type: 'LAINNYA', notes: '' })
   389→      setNamesText('')
   390→      // Reload events list
   391→      api('/api/events').then(setEvents).catch(() => {})
   392→      setMode('scheduled')
   393→    } catch (e: any) {
   394→      addToast(e.message, 'error')
   395→    } finally {
   396→      setSubmitting(false)
   397→    }
   398→  }
   399→
   400→  // === Submit Manual Names (for existing event) ===
   401→  const handleManualNamesSubmit = async () => {
   402→    if (!selectedEvent) {
   403→      addToast('Pilih event dulu', 'error')
   404→      return
   405→    }
   406→    const names = namesText.split('\n').map((n) => n.trim()).filter((n) => n.length >= 2)
   407→    if (names.length === 0) {
   408→      addToast('Daftar nama wajib diisi (minimal 1 nama, satu per baris)', 'error')
   409→      return
   410→    }
   411→
   412→    setSubmitting(true)
   413→    try {
   414→      const res = await fetch('/api/attendance/bulk', {
   415→        method: 'POST',
   416→        headers: { 'Content-Type': 'application/json', 'x-user-id': user?.id || '' },
   417→        body: JSON.stringify({
   418→          mode: 'manual_names',
   419→          eventId: selectedEvent,
   420→          namesText,
   421→          status: 'PRESENT',
   422→        }),
   423→      })
   424→      const data = await res.json()
   425→      if (!res.ok || !data.success) throw new Error(data.error)
   426→
   427→      addToast(data.message, 'success')
   428→      setNamesText('')
   429→      // Reload attendance
   430→      api(`/api/attendance?eventId=${selectedEvent}`).then(setAttendance).catch(() => {})
   431→    } catch (e: any) {
   432→      addToast(e.message, 'error')
   433→    } finally {
   434→      setSubmitting(false)
   435→    }
   436→  }
   437→
   438→  // === Parse CSV file client-side ===
   439→  const handleCsvFileChange = async (file: File | null) => {
   440→    if (!file) {
   441→      setCsvFile(null)
   442→      setCsvPreview([])
   443→      return
   444→    }
   445→    setCsvFile(file)
   446→    try {
   447→      const text = await file.text()
   448→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   449→      const rows = lines.map((line) => {
   450→        // Parse CSV (handle quoted values)
   451→        const cols = line.match(/("([^"]*)"|([^,]*)),?/g)?.map((c) => c.replace(/,$/, '').replace(/^"|"$/g, '')) || []
   452→        return {
   453→          name: cols[0] || '',
   454→          phone: cols[1] || '',
   455→          territoryName: cols[2] || '',
   456→        }
   457→      }).filter((r) => r.name.length >= 2)
   458→      setCsvPreview(rows.slice(0, 10)) // Preview first 10
   459→    } catch (e: any) {
   460→      addToast(`Gagal parse CSV: ${e.message}`, 'error')
   461→    }
   462→  }
   463→
   464→  // === Submit CSV Upload ===
   465→  const handleCsvSubmit = async () => {
   466→    if (!selectedEvent) {
   467→      addToast('Pilih event dulu', 'error')
   468→      return
   469→    }
   470→    if (!csvFile || csvPreview.length === 0) {
   471→      addToast('Pilih file CSV dulu', 'error')
   472→      return
   473→    }
   474→
   475→    setSubmitting(true)
   476→    try {
   477→      // Parse full CSV (not just preview)
   478→      const text = await csvFile.text()
   479→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   480→      const rows = lines.map((line) => {
   481→        const cols = line.match(/("([^"]*)"|([^,]*)),?/g)?.map((c) => c.replace(/,$/, '').replace(/^"|"$/g, '')) || []
   482→        return {
   483→          name: cols[0] || '',
   484→          phone: cols[1] || '',
   485→          territoryName: cols[2] || '',
   486→        }
   487→      }).filter((r) => r.name.length >= 2)
   488→
   489→      const res = await fetch('/api/attendance/bulk', {
   490→        method: 'POST',
   491→        headers: { 'Content-Type': 'application/json', 'x-user-id': user?.id || '' },
   492→        body: JSON.stringify({
   493→          mode: 'csv_upload',
   494→          eventId: selectedEvent,
   495→          rows,
   496→        }),
   497→      })
   498→      const data = await res.json()
   499→      if (!res.ok || !data.success) throw new Error(data.error)
   500→
   501→      addToast(data.message, 'success')
   502→      setCsvFile(null)
   503→      setCsvPreview([])
   504→      api(`/api/attendance?eventId=${selectedEvent}`).then(setAttendance).catch(() => {})
   505→    } catch (e: any) {
   506→      addToast(e.message, 'error')
   507→    } finally {
   508→      setSubmitting(false)
   509→    }
   510→  }
   511→
   512→  if (loading) return <LoadingState />
   513→
   514→  const getMemberStatus = (memberId: string) => {
   515→    const rec = attendance.find((a) => a.memberId === memberId)
   516→    return rec?.status || null
   517→  }
   518→
   519→  // Stats for selected event
   520→  const presentCount = attendance.filter((a) => a.status === 'PRESENT').length
   521→  const absentCount = attendance.filter((a) => a.status === 'ABSENT').length
   522→  const excusedCount = attendance.filter((a) => a.status === 'EXCUSED').length
   523→
   524→  return (
   525→    <div className="space-y-4">
   526→      {/* Mode selector */}
