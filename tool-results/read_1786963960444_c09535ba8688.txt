   143→function formatFileSize(bytes?: number): string {
   144→  if (!bytes) return '-'
   145→  if (bytes < 1024) return `${bytes} B`
   146→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   147→  return `${(bytes / 1024 / 1024).toFixed(2)} MB`
   148→}
   149→
   150→function getFileIcon(fileType?: string | null) {
   151→  if (!fileType) return FileText
   152→  const t = fileType.toUpperCase()
   153→  if (['JPG', 'JPEG', 'PNG', 'WEBP', 'GIF'].includes(t)) return ImageIcon
   154→  if (['MP4', 'MOV', 'WEBM'].includes(t)) return Video
   155→  if (['PDF'].includes(t)) return FileText
   156→  if (['DOC', 'DOCX', 'TXT'].includes(t)) return FileText
   157→  if (['XLS', 'XLSX', 'CSV'].includes(t)) return FileCheck
   158→  return FileText
   159→}
   160→
   161→function validateFile(file: File): { valid: boolean; error?: string } {
   162→  if (file.size > MAX_FILE_SIZE) {
   163→    return { valid: false, error: `Ukuran > 4MB (${(file.size / 1024 / 1024).toFixed(2)}MB)` }
   164→  }
   165→  const parts = file.name.toLowerCase().split('.')
   166→  const ext = parts.length > 1 ? parts[parts.length - 1] : ''
   167→  if (!ALLOWED_EXT_ARRAY.includes(ext)) {
   168→    return { valid: false, error: `Format .${ext} tidak didukung` }
   169→  }
   170→  return { valid: true }
   171→}

  1275→function BulkUploadDialog({
  1276→  open, onOpenChange, category, level, territoryCode, territoryName, accentColor, onSuccess,
  1277→}: {

  1279→  onOpenChange: (o: boolean) => void
  1280→  category: Category
  1281→  level: Level
  1282→  territoryCode: string
  1283→  territoryName: string
  1284→  accentColor: string
  1285→  onSuccess: () => void
  1286→}) {
  1287→  const addToast = useToastStore((s) => s.addToast)
  1288→  const user = useAuthStore((s) => s.user)
  1289→  const [files, setFiles] = useState<File[]>([])
  1290→  const [filesError, setFilesError] = useState<string | null>(null)
  1291→  const [isDragging, setIsDragging] = useState(false)
  1292→  const [titlePrefix, setTitlePrefix] = useState('')
  1293→  const [location, setLocation] = useState('')
  1294→  const [eventDate, setEventDate] = useState('')
  1295→  const [status, setStatus] = useState<Status>('DIRENCANAKAN')
  1296→  const [saving, setSaving] = useState(false)
  1297→  const fileInputRef = useRef<HTMLInputElement>(null)
  1298→
  1299→  useEffect(() => {
  1300→    if (open) {
  1301→      setFiles([])
  1302→      setFilesError(null)
  1303→      setTitlePrefix('')
  1304→      setLocation('')
  1305→      setEventDate('')
  1306→      setStatus('DIRENCANAKAN')
  1307→      if (fileInputRef.current) fileInputRef.current.value = ''
  1308→    }
  1309→  }, [open])
  1310→
  1311→  const handleFilesSelect = (fileList: FileList | null) => {
  1312→    setFilesError(null)
  1313→    if (!fileList || fileList.length === 0) return
  1314→    const arr = Array.from(fileList)
  1315→    if (arr.length > 10) {
  1316→      setFilesError('Maksimal 10 file per batch')
  1317→      return
  1318→    }
  1319→    const invalid = arr.find(f => {
  1320→      const v = validateFile(f)
  1321→      return !v.valid
  1322→    })
  1323→    if (invalid) {
  1324→      const v = validateFile(invalid)
  1325→      setFilesError(`File "${invalid.name}": ${v.error}`)
  1326→      return
  1327→    }
  1328→    setFiles(arr)
  1329→  }
  1330→
  1331→  const handleSubmit = async () => {
  1332→    if (files.length === 0) { addToast('Tidak ada file', 'error'); return }
  1333→    if (filesError) { addToast(filesError, 'error'); return }
  1334→
  1335→    setSaving(true)
  1336→    try {
  1337→      const formData = new FormData()
  1338→      files.forEach(f => formData.append('files', f))
  1339→      formData.append('titlePrefix', titlePrefix)
  1340→      formData.append('category', category)
  1341→      formData.append('level', level)
  1342→      formData.append('territoryCode', territoryCode)
  1343→      formData.append('territoryName', territoryName)
  1344→      formData.append('location', location)
  1345→      formData.append('eventDate', eventDate)
  1346→      formData.append('status', status)
  1347→
  1348→      const res = await fetch('/api/program-documents/bulk', {
  1349→        method: 'POST',
  1350→        headers: { 'x-user-id': user?.id || '' },
  1351→        body: formData,
  1352→      })
  1353→      const data = await res.json()

  1355→        throw new Error(data.error || `HTTP ${res.status}`)
  1356→      }
  1357→      const { uploaded, failed } = data.data
  1358→      addToast(`Upload selesai: ${uploaded.length} berhasil, ${failed.length} gagal`, failed.length > 0 ? 'warning' : 'success')
  1359→      if (failed.length > 0) {
  1360→        console.warn('Failed uploads:', failed)
  1361→      }
  1362→      onSuccess()
  1363→    } catch (e: any) {
  1364→      addToast(`Gagal bulk upload: ${e.message}`, 'error')
  1365→    } finally {
  1366→      setSaving(false)
  1367→    }
  1368→  }

  1372→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
  1373→        <DialogHeader>
  1374→          <DialogTitle>Bulk Upload — {level} {territoryName}</DialogTitle>
  1375→          <DialogDescription>
  1376→            Upload multiple file sekaligus (maksimal 10 file). Semua file akan diberi metadata yang sama (lokasi, tanggal, status). Judul otomatis pakai nama file atau prefix + nomor urut.
  1377→          </DialogDescription>
  1378→        </DialogHeader>
  1379→        <div className="space-y-3 py-2">
  1380→          {/* Dropzone */}
  1381→          <div className="space-y-1.5">
  1382→            <Label>Pilih Multiple File (maks 10)</Label>
  1383→            <div
  1384→              onClick={() => fileInputRef.current?.click()}
  1385→              onDragOver={(e) => { e.preventDefault(); setIsDragging(true) }}
  1386→              onDragLeave={() => setIsDragging(false)}
  1387→              onDrop={(e) => {
  1388→                e.preventDefault()
  1389→                setIsDragging(false)
  1390→                handleFilesSelect(e.dataTransfer.files)
  1391→              }}
  1392→              className={`border-2 border-dashed rounded-lg p-6 text-center cursor-pointer transition-colors ${
  1393→                isDragging ? 'border-orange-500 bg-orange-50' : 'hover:bg-accent'

  1489→function EditDialog({
  1490→  doc, open, onOpenChange, onSuccess,
  1491→}: {
  1492→  doc: ProgramDoc
  1493→  open: boolean
  1494→  onOpenChange: (o: boolean) => void
  1495→  onSuccess: () => void
  1496→}) {
  1497→  const addToast = useToastStore((s) => s.addToast)
  1498→  const user = useAuthStore((s) => s.user)
  1499→  const [form, setForm] = useState({
  1500→    title: doc.title,
  1501→    description: doc.description || '',
  1502→    location: doc.location || '',
  1503→    eventDate: doc.eventDate ? doc.eventDate.substring(0, 10) : '',
  1504→    status: doc.status,
  1505→  })
  1506→  const [replaceFile, setReplaceFile] = useState(false)
  1507→  const [newFile, setNewFile] = useState<File | null>(null)
  1508→  const [fileError, setFileError] = useState<string | null>(null)
  1509→  const [saving, setSaving] = useState(false)
  1510→  const fileInputRef = useRef<HTMLInputElement>(null)
  1511→
  1512→  const handleFileSelect = (f: File | null) => {
  1513→    setFileError(null)
  1514→    if (!f) { setNewFile(null); return }
  1515→    const v = validateFile(f)
  1516→    if (!v.valid) { setFileError(v.error || 'File tidak valid'); setNewFile(null); return }
  1517→    setNewFile(f)
  1518→  }
  1519→
  1520→  const handleSubmit = async () => {
  1521→    if (!form.title.trim()) { addToast('Judul wajib diisi', 'error'); return }
  1522→    if (replaceFile && !newFile) { addToast('Pilih file baru atau uncheck "Ganti file"', 'error'); return }
  1523→    if (fileError) { addToast(fileError, 'error'); return }
  1524→
  1525→    setSaving(true)
  1526→    try {
  1527→      if (replaceFile && newFile) {
  1528→        const formData = new FormData()
  1529→        formData.append('file', newFile)
  1530→        formData.append('title', form.title)
  1531→        formData.append('description', form.description)
  1532→        formData.append('location', form.location)
  1533→        formData.append('eventDate', form.eventDate)
  1534→        formData.append('status', form.status)
  1535→
  1536→        const res = await fetch(`/api/program-documents/${doc.id}`, {
  1537→          method: 'PUT',
  1538→          headers: { 'x-user-id': user?.id || '' },
  1539→          body: formData,
  1540→        })
  1541→        const data = await res.json()
  1542→        if (!res.ok || !data.success) throw new Error(data.error || `HTTP ${res.status}`)
  1543→        addToast('File & metadata berhasil diperbarui', 'success')
  1544→      } else {
  1545→        await api(`/api/program-documents/${doc.id}`, {
  1546→          method: 'PUT',
  1547→          body: JSON.stringify({
  1548→            title: form.title,
  1549→            description: form.description,
  1550→            location: form.location,
  1551→            eventDate: form.eventDate,
  1552→            status: form.status,
  1553→          }),
  1554→        })
  1555→        addToast('Metadata berhasil diperbarui', 'success')
  1556→      }
  1557→      onSuccess()
  1558→    } catch (e: any) {
  1559→      addToast(`Gagal update: ${e.message}`, 'error')
  1560→    } finally {
  1561→      setSaving(false)
  1562→    }
  1563→  }
  1564→
  1565→  return (
  1566→    <Dialog open={open} onOpenChange={onOpenChange}>
  1567→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
  1568→        <DialogHeader>
  1569→          <DialogTitle>Edit Dokumen — {doc.title}</DialogTitle>
  1570→          <DialogDescription>
  1571→            Level: {doc.level || '-'} • Wilayah: {doc.territoryName || '-'}
  1572→            {doc.fileName && (<> • File: <strong>{doc.fileName}</strong></>)}
  1573→          </DialogDescription>
  1574→        </DialogHeader>
  1575→        <div className="space-y-3 py-2">
  1576→          {doc.fileName && (
  1577→            <div className="space-y-2">
  1578→              <label className="flex items-center gap-2 cursor-pointer">
  1579→                <Checkbox
  1580→                  checked={replaceFile}
  1581→                  onCheckedChange={(checked) => { setReplaceFile(checked === true); setNewFile(null); setFileError(null) }}
  1582→                />
  1583→                <span className="text-sm font-medium">Ganti file bukti</span>
  1584→              </label>
  1585→              {replaceFile && (
  1586→                <div
  1587→                  onClick={() => fileInputRef.current?.click()}
  1588→                  onDragOver={(e) => { e.preventDefault() }}
  1589→                  onDrop={(e) => {
  1590→                    e.preventDefault()
  1591→                    handleFileSelect(e.dataTransfer.files?.[0] || null)
  1592→                  }}
  1593→                  className={`border-2 border-dashed rounded-lg p-4 text-center cursor-pointer hover:bg-accent ${fileError ? 'border-red-400' : ''}`}
  1594→                >
  1595→                  {newFile ? (
  1596→                    <div className="flex items-center justify-center gap-2 text-sm">
  1597→                      <FileText className="w-4 h-4 text-orange-600" />
  1598→                      <span>{newFile.name}</span>
  1599→                      <span className="text-muted-foreground">({formatFileSize(newFile.size)})</span>
  1600→                      <Button type="button" variant="ghost" size="sm" onClick={(e) => { e.stopPropagation(); setNewFile(null); if (fileInputRef.current) fileInputRef.current.value = '' }} className="h-6 w-6 p-0">
  1601→                        <X className="w-3 h-3" />
  1602→                      </Button>
  1603→                    </div>
  1604→                  ) : (
  1605→                    <div className="text-sm text-muted-foreground">Klik atau drag file baru</div>
  1606→                  )}
  1607→                </div>
  1608→              )}
  1609→              <input
  1610→                ref={fileInputRef}
  1611→                type="file"
  1612→                accept={ACCEPTED_EXTENSIONS}
  1613→                onChange={(e) => handleFileSelect(e.target.files?.[0] || null)}
  1614→                className="hidden"
  1615→              />
  1616→              {fileError && <p className="text-xs text-red-600">{fileError}</p>}
  1617→            </div>
  1618→          )}
  1619→
  1620→          <div className="space-y-1.5">
  1621→            <Label>Judul *</Label>
  1622→            <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} />
  1623→          </div>
  1624→
  1625→          <div className="space-y-1.5">
  1626→            <Label>Deskripsi</Label>
  1627→            <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={3} />
  1628→          </div>
  1629→
  1630→          <div className="grid grid-cols-2 gap-3">
  1631→            <div className="space-y-1.5">
  1632→              <Label>Lokasi</Label>
  1633→              <Input value={form.location} onChange={(e) => setForm({ ...form, location: e.target.value })} />
  1634→            </div>
  1635→            <div className="space-y-1.5">
  1636→              <Label>Tanggal</Label>
  1637→              <Input type="date" value={form.eventDate} onChange={(e) => setForm({ ...form, eventDate: e.target.value })} />
  1638→            </div>
  1639→          </div>
  1640→
  1641→          <div className="space-y-1.5">
  1642→            <Label>Status</Label>
  1643→            <Select value={form.status} onValueChange={(v) => setForm({ ...form, status: v as Status })}>
  1644→              <SelectTrigger>
  1645→                <SelectValue />
  1646→              </SelectTrigger>
  1647→              <SelectContent>
  1648→                <SelectItem value="DIRENCANAKAN">Direncanakan</SelectItem>
  1649→                <SelectItem value="BERJALAN">Berjalan</SelectItem>
  1650→                <SelectItem value="SELESAI">Selesai</SelectItem>
  1651→                <SelectItem value="DITUNDA">Ditunda</SelectItem>
  1652→              </SelectContent>
  1653→            </Select>
  1654→          </div>
  1655→        </div>
  1656→
  1657→        <DialogFooter>
  1658→          <Button variant="outline" onClick={() => onOpenChange(false)}>Batal</Button>
  1659→          <Button onClick={handleSubmit} disabled={saving} className="gap-1">
  1660→            {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Edit className="w-4 h-4" />}
  1661→            Simpan Perubahan
  1662→          </Button>
  1663→        </DialogFooter>
  1664→      </DialogContent>
  1665→    </Dialog>
  1666→  )
  1667→}
