   363→   363→function ProgramLevelView({
   364→   364→  title, description, icon: Icon, accentColor, items, loading, category, level, territoryCode, territoryName, onBack, setItems,
   365→   365→}: {
   366→   366→  title: string
   367→   367→  description: string
   368→   368→  icon: any
   369→   369→  accentColor: string
   370→   370→  items: ProgramDoc[]
   371→   371→  loading: boolean
   372→   372→  category: string
   373→   373→  level: Level
   374→   374→  territoryCode: string
   375→   375→  territoryName: string
   376→   376→  onBack: () => void
   377→   377→  setItems: React.Dispatch<React.SetStateAction<ProgramDoc[]>>
   378→   378→}) {
   379→   379→  const addToast = useToastStore((s) => s.addToast)
   380→   380→  const user = useAuthStore((s) => s.user)
   381→   381→  const isSuperAdmin = useIsSuperAdmin()
   382→   382→  const [search, setSearch] = useState('')
   383→   383→  const [uploadOpen, setUploadOpen] = useState(false)
   384→   384→  const [editDoc, setEditDoc] = useState<ProgramDoc | null>(null)
   385→   385→  const [deleteDoc, setDeleteDoc] = useState<ProgramDoc | null>(null)
   386→   386→  const [viewingDoc, setViewingDoc] = useState<ProgramDoc | null>(null)
   387→   387→
   388→   388→  // === Helper: format ukuran file ===
   389→   389→  const formatFileSize = (bytes?: number) => {
   390→   390→    if (!bytes) return '-'
   391→   391→    if (bytes < 1024) return `${bytes} B`
