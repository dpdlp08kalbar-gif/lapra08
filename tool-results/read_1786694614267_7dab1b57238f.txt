    29→export function ProgramContentManager({ title, description, icon: Icon, category, accentColor }: {
    30→  title: string; description: string; icon: any; category: string; accentColor: string
    31→}) {
    32→  const [view, setView] = useState<'home' | 'dpn' | 'dpd-list' | 'dpd-detail' | 'dpc-list' | 'dpc-detail'>('home')
    33→  const [territories, setTerritories] = useState<Territory[]>([])
    34→  const [regencies, setRegencies] = useState<Territory[]>([])
    35→  const [selectedProv, setSelectedProv] = useState<Territory | null>(null)
    36→  const [selectedRegency, setSelectedRegency] = useState<Territory | null>(null)
    37→  const [items, setItems] = useState<any[]>([])
    38→  const [loading, setLoading] = useState(true)
    39→
    40→  useEffect(() => {
    41→    Promise.all([
    42→      api('/api/territory'),
    43→      api('/api/gallery'),
    44→    ]).then(([allTerr, allGallery]) => {
    45→      setTerritories((allTerr as any[]).filter(t => t.level === 'PROVINCE'))
    46→      setRegencies((allTerr as any[]).filter(t => t.level === 'REGENCY'))
    47→      setItems((allGallery as any[]).filter(a => a.category === category))
    48→    }).catch(() => {}).finally(() => setLoading(false))
    49→  }, [])
    50→
    51→  const isFiltered = (i: any, code: string, level: string) =>
    52→    i.territoryCode === code || (i.level === level && i.territoryCode === code)
    53→
    54→  const dpnItems = items.filter(i => !i.territoryCode || i.territoryCode === 'ID' || i.level === 'DPN')
    55→  const dpdCount = items.filter(i => i.level === 'DPD' || (i.territoryCode && i.territoryCode.length === 2 && i.territoryCode !== 'ID')).length
    56→  const dpcCount = items.filter(i => i.level === 'DPC' || (i.territoryCode && i.territoryCode.length >= 4)).length
    57→
    58→  // === HOME: 3 Kartu DPN/DPD/DPC ===
    59→  if (view === 'home') {
    60→    const cards = [
    61→      { key: 'dpn', title: 'DPN', subtitle: 'Pusat Nasional', desc: `${title} tingkat DPN`, count: dpnItems.length, icon: Crown, grad: 'from-red-500 to-orange-600' },
    62→      { key: 'dpd', title: 'DPD', subtitle: 'Provinsi', desc: `${title} DPD se-Indonesia + LN`, count: dpdCount, icon: Building2, grad: 'from-blue-500 to-cyan-600' },
    63→      { key: 'dpc', title: 'DPC', subtitle: 'Kabupaten/Kota', desc: `${title} DPC per DPD`, count: dpcCount, icon: MapPin, grad: 'from-emerald-500 to-teal-600' },
    64→    ]
    65→    return (
    66→      <div className="space-y-4">
    67→        <div className="rounded-lg bg-blue-50 border border-blue-200 p-3 text-xs text-blue-800">
    68→          <strong>Hierarki:</strong> DPN (Pusat Nasional) → DPD (Provinsi) → DPC (Kabupaten/Kota).
    69→          Pilih tingkat untuk melihat {title.toLowerCase()} masing-masing wilayah.
    70→        </div>
    71→        <div className="grid gap-4 md:grid-cols-3">
    72→          {cards.map(c => (
    73→            <div key={c.key} className="rounded-2xl border-2 hover:shadow-lg transition-all cursor-pointer overflow-hidden"
    74→              onClick={() => { setView(c.key === 'dpn' ? 'dpn' : c.key === 'dpd' ? 'dpd-list' : 'dpc-list'); setSelectedProv(null); setSelectedRegency(null) }}>
    75→              <div className={`bg-gradient-to-br ${c.grad} p-5 text-white`}>
    76→                <c.icon className="w-8 h-8 mb-2" />
    77→                <div className="text-xl font-bold">{c.title}</div>
    78→                <div className="text-sm opacity-90">{c.subtitle}</div>
    79→              </div>
    80→              <div className="p-4 bg-white">
    81→                <div className="text-sm text-muted-foreground mb-2">{c.desc}</div>
    82→                <div className="flex items-center justify-between">
    83→                  <Badge variant="outline" className="text-[13px]">{c.count} program</Badge>
    84→                  <span className="text-xs font-medium text-blue-600">Buka <ChevronRight className="w-4 h-4 inline" /></span>
    85→                </div>
    86→              </div>
    87→            </div>
    88→          ))}
    89→        </div>
    90→      </div>
    91→    )
    92→  }
    93→
    94→  // === DPN VIEW ===
    95→  if (view === 'dpn') {
    96→    return <ProgramLevelView title={`${title} — DPN (Pusat Nasional)`} description={description} icon={Icon} accentColor={accentColor} items={dpnItems} loading={loading} category={category} level="DPN" territoryCode="ID" territoryName="DPN (Pusat Nasional)" onBack={() => setView('home')} items_ref={items} setItems={setItems} />
    97→  }
    98→
    99→  // === DPD LIST VIEW ===
   100→  if (view === 'dpd-list') {
   101→    return (
   102→      <div className="space-y-4">
   103→        <Button variant="ghost" size="sm" onClick={() => setView('home')}><ChevronRight className="w-4 h-4 rotate-180" /> Kembali</Button>
   104→        <h3 className="text-base font-bold flex items-center gap-2"><Building2 className="w-5 h-5 text-blue-600" /> Pilih DPD (Provinsi)</h3>
   105→        <div className="grid gap-2 md:grid-cols-2 lg:grid-cols-3">
   106→          {territories.map(prov => {
   107→            const count = items.filter(i => isFiltered(i, prov.code, 'DPD')).length
   108→            return (
   109→              <button key={prov.code} onClick={() => { setSelectedProv(prov); setView('dpd-detail') }}
   110→                className="flex items-center justify-between p-3 rounded-lg border hover:bg-accent transition-all text-left">
   111→                <span className="text-sm font-medium">{prov.name}</span>
   112→                <Badge variant="outline" className="text-[13px]">{count > 0 ? `${count} program` : 'Kosong'}</Badge>
   113→              </button>
   114→            )
   115→          })}
   116→        </div>
   117→      </div>
   118→    )
   119→  }
   120→
   121→  // === DPD DETAIL VIEW ===
   122→  if (view === 'dpd-detail' && selectedProv) {
   123→    const dpdItems = items.filter(i => isFiltered(i, selectedProv.code, 'DPD'))
   124→    return <ProgramLevelView title={`${title} — DPD ${selectedProv.name}`} description={description} icon={Icon} accentColor={accentColor} items={dpdItems} loading={loading} category={category} level="DPD" territoryCode={selectedProv.code} territoryName={`DPD ${selectedProv.name}`} onBack={() => { setSelectedProv(null); setView('dpd-list') }} items_ref={items} setItems={setItems} />
   125→  }
   126→
   127→  // === DPC LIST VIEW (pilih provinsi dulu) ===
   128→  if (view === 'dpc-list') {
   129→    return (
   130→      <div className="space-y-4">
   131→        <Button variant="ghost" size="sm" onClick={() => setView('home')}><ChevronRight className="w-4 h-4 rotate-180" /> Kembali</Button>
   132→        <h3 className="text-base font-bold flex items-center gap-2"><MapPin className="w-5 h-5 text-emerald-600" /> Pilih DPD (Provinsi) untuk lihat DPC</h3>
   133→        <div className="grid gap-2 md:grid-cols-2 lg:grid-cols-3">
   134→          {territories.map(prov => {
   135→            const regCount = regencies.filter(r => r.parentId === prov.id).length
   136→            return (
   137→              <button key={prov.code} onClick={() => { setSelectedProv(prov); setView('dpc-detail'); setSelectedRegency(null) }}
   138→                className="flex items-center justify-between p-3 rounded-lg border hover:bg-accent transition-all text-left">
   139→                <span className="text-sm font-medium">{prov.name}</span>
   140→                <Badge variant="outline" className="text-[13px]">{regCount > 0 ? `${regCount} DPC` : 'Kosong'}</Badge>
   141→              </button>
   142→            )
   143→          })}
   144→        </div>
   145→      </div>
   146→    )
   147→  }
   148→
   149→  // === DPC DETAIL VIEW (list kab/kota di provinsi terpilih) ===
   150→  if (view === 'dpc-detail' && selectedProv) {
   151→    const provRegencies = regencies.filter(r => r.parentId === selectedProv.id)
   152→    return (
   153→      <div className="space-y-4">
   154→        <Button variant="ghost" size="sm" onClick={() => { setSelectedProv(null); setView('dpc-list') }}><ChevronRight className="w-4 h-4 rotate-180" /> Kembali ke daftar provinsi</Button>
   155→        <h3 className="text-base font-bold flex items-center gap-2"><MapPin className="w-5 h-5 text-emerald-600" /> DPC di {selectedProv.name}</h3>
   156→        <div className="grid gap-2 md:grid-cols-2 lg:grid-cols-3">
   157→          {provRegencies.length === 0 ? (
   158→            <p className="text-sm text-muted-foreground col-span-full">Belum ada DPC terdaftar di provinsi ini.</p>
   159→          ) : provRegencies.map(reg => {
   160→            const count = items.filter(i => isFiltered(i, reg.code, 'DPC')).length
   161→            return (
   162→              <button key={reg.code} onClick={() => setSelectedRegency(reg)}
   163→                className="flex items-center justify-between p-3 rounded-lg border hover:bg-accent transition-all text-left">
   164→                <span className="text-sm font-medium">{reg.name}</span>
   165→                <Badge variant="outline" className="text-[13px]">{count > 0 ? `${count} program` : 'Kosong'}</Badge>
   166→              </button>
   167→            )
   168→          })}
   169→        </div>
   170→        {selectedRegency && (() => {
   171→          const dpcItems = items.filter(i => isFiltered(i, selectedRegency.code, 'DPC'))
   172→          return <ProgramLevelView title={`${title} — DPC ${selectedRegency.name}`} description={description} icon={Icon} accentColor={accentColor} items={dpcItems} loading={loading} category={category} level="DPC" territoryCode={selectedRegency.code} territoryName={`DPC ${selectedRegency.name}`} onBack={() => setSelectedRegency(null)} items_ref={items} setItems={setItems} />
   173→        })()}
   174→      </div>
   175→    )
   176→  }
   177→
   178→  return null
   179→}

   182→// PROGRAM LEVEL VIEW — Content for DPN/DPD/DPC
   183→// ============================================================
   184→function ProgramLevelView({
   185→  title, description, icon: Icon, accentColor, items, loading, category, level, territoryCode, territoryName, onBack, items_ref, setItems,
   186→}: any) {
   187→  const addToast = useToastStore((s) => s.addToast)
   188→  const isSuperAdmin = useIsSuperAdmin()
   189→  const [search, setSearch] = useState('')
   190→  const [addOpen, setAddOpen] = useState(false)
   191→  const [editItem, setEditItem] = useState<any>(null)
   192→  const [deleteItem, setDeleteItem] = useState<any>(null)
   193→  const [form, setForm] = useState({ title: '', description: '', location: '', date: '', status: 'DIRENCANAKAN' })
   194→  const [saving, setSaving] = useState(false)
   195→  const [pdfUploadOpen, setPdfUploadOpen] = useState(false)
   196→  const [pdfFile, setPdfFile] = useState<File | null>(null)
   197→  const [ocrLoading, setOcrLoading] = useState(false)
   198→  const [extractResult, setExtractResult] = useState<any>(null)
   199→  const [filterStatus, setFilterStatus] = useState('ALL')
   200→  // Bukti pelaksanaan state
   201→  const [evidenceDialog, setEvidenceDialog] = useState<any>(null) // { item, files: [] }
   202→  const [evidenceFile, setEvidenceFile] = useState<File | null>(null)
   203→  const [evidenceLoading, setEvidenceLoading] = useState(false)
   204→
   205→  const reload = () => {
   206→    api('/api/gallery').then((all: any[]) => {
   207→      setItems(all.filter(a => a.category === category))
   208→    }).catch(() => {})
   209→  }
   210→
   211→  const handleSave = async (e: React.FormEvent) => {
   212→    e.preventDefault()
   213→    const itemData = {
   214→      id: editItem?.id || `prog_${Date.now()}`,
   215→      title: form.title, description: form.description, location: form.location,
   216→      date: form.date, status: form.status, category, level, territoryCode, territoryName,
   217→    }
   218→    try {
   219→      if (editItem) {
   220→        await api('/api/gallery', { method: 'PUT', body: JSON.stringify({ id: editItem.id, ...itemData }) })
   221→        addToast('Program diperbarui', 'success')
   222→      } else {
   223→        await api('/api/gallery', { method: 'POST', body: JSON.stringify(itemData) })
   224→        addToast('Program ditambahkan', 'success')
   225→      }
   226→      setForm({ title: '', description: '', location: '', date: '', status: 'DIRENCANAKAN' })
   227→      setAddOpen(false); setEditItem(null); reload()
   228→    } catch (e: any) { addToast(e.message, 'error') }
   229→  }
   230→
   231→  const handleDelete = async () => {
   232→    if (!deleteItem) return
   233→    try {
   234→      await api(`/api/gallery?id=${deleteItem.id}`, { method: 'DELETE' })
   235→      addToast('Program dihapus', 'success')
   236→      setDeleteItem(null); reload()
   237→    } catch (e: any) { addToast(e.message, 'error') }
   238→  }
   239→
   240→  // === Upload Bukti Pelaksanaan ===
   241→  const handleUploadEvidence = async () => {
   242→    if (!evidenceFile || !evidenceDialog?.item) return
   243→    const max = 5 * 1024 * 1024
   244→    if (evidenceFile.size > max) { addToast('Ukuran file maksimal 5MB', 'error'); return }
   245→
   246→    setEvidenceLoading(true)
   247→    try {
   248→      const buf = Buffer.from(await evidenceFile.arrayBuffer())
   249→      const ext = evidenceFile.name.toLowerCase().match(/\.([^.]+)$/)?.[1] || 'jpg'
   250→      const mime = ext === 'pdf' ? 'application/pdf'
   251→        : ext === 'png' ? 'image/png'
   252→        : ext === 'jpg' || ext === 'jpeg' ? 'image/jpeg'
   253→        : ext === 'doc' || ext === 'docx' ? 'application/msword'
   254→        : 'application/octet-stream'
   255→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   256→
   257→      // Get existing evidence files
   258→      const item = evidenceDialog.item
   259→      const existingEvidence = item.evidenceFiles ? (typeof item.evidenceFiles === 'string' ? JSON.parse(item.evidenceFiles) : item.evidenceFiles) : []
   260→      const newEvidence = [...existingEvidence, {
   261→        id: `ev_${Date.now()}`,
   262→        fileName: evidenceFile.name,
   263→        fileSize: evidenceFile.size,
   264→        fileType: ext,
   265→        dataUrl,
   266→        uploadedAt: new Date().toISOString(),
   267→      }]
   268→
   269→      // Update program item with evidence + auto-update status to BERJALAN if still DIRENCANAKAN
   270→      const updateData: any = { evidenceFiles: JSON.stringify(newEvidence) }
   271→      if ((item.status || 'DIRENCANAKAN') === 'DIRENCANAKAN') {
   272→        updateData.status = 'BERJALAN'
   273→      }
   274→      await api('/api/gallery', { method: 'PUT', body: JSON.stringify({ id: item.id, ...updateData }) })
   275→
   276→      addToast(`Bukti pelaksanaan "${evidenceFile.name}" berhasil diupload${updateData.status ? '. Status diubah ke BERJALAN' : ''}`, 'success')
   277→      setEvidenceFile(null)
   278→      setEvidenceDialog(null)
   279→      reload()
   280→    } catch (e: any) { addToast(e.message, 'error') }
   281→    finally { setEvidenceLoading(false) }
   282→  }
   283→
   284→  // === Delete evidence file ===
   285→  const handleDeleteEvidence = async (itemId: string, evidenceId: string, currentEvidence: any[]) => {
   286→    try {
   287→      const filtered = currentEvidence.filter(e => e.id !== evidenceId)
   288→      await api('/api/gallery', { method: 'PUT', body: JSON.stringify({ id: itemId, evidenceFiles: JSON.stringify(filtered) }) })
   289→      addToast('Bukti dihapus', 'success')
   290→      setEvidenceDialog(null)
   291→      reload()
   292→    } catch (e: any) { addToast(e.message, 'error') }
   293→  }
   294→
   295→  if (loading) return <LoadingState />
   296→
   297→  const filtered = items.filter((i: any) =>
   298→    (!search || i.title?.toLowerCase().includes(search.toLowerCase()) || i.description?.toLowerCase().includes(search.toLowerCase())) &&
   299→    (filterStatus === 'ALL' || (i.status || 'DIRENCANAKAN') === filterStatus)
   300→  )
   301→
   302→  // Stats by status
   303→  const statusCounts = {
   304→    DIRENCANAKAN: items.filter(i => (i.status || 'DIRENCANAKAN') === 'DIRENCANAKAN').length,
   305→    BERJALAN: items.filter(i => i.status === 'BERJALAN').length,
   306→    SELESAI: items.filter(i => i.status === 'SELESAI').length,
   307→    DITUNDA: items.filter(i => i.status === 'DITUNDA').length,
   308→  }
   309→
   310→  return (
   311→    <div className="space-y-4">
   312→      <Button variant="ghost" size="sm" onClick={onBack}><ChevronRight className="w-4 h-4 rotate-180" /> Kembali</Button>
   313→      <div className="flex items-center justify-between flex-wrap gap-2">
   314→        <h3 className="text-base font-bold flex items-center gap-2">
   315→          <Icon className={`w-5 h-5 bg-gradient-to-br ${accentColor} bg-clip-text`} />
   316→          {title}
   317→          <Badge variant="outline" className="text-[13px]">{filtered.length} program</Badge>
   318→        </h3>
   319→        {isSuperAdmin && (
   320→          <div className="flex gap-2">
   321→            <Button size="sm" variant="outline" onClick={() => setPdfUploadOpen(true)}>
   322→              <Upload className="w-4 h-4 mr-1" /> Upload PDF
   323→            </Button>
   324→            <Button size="sm" onClick={() => { setEditItem(null); setForm({ title: '', description: '', location: '', date: '', status: 'DIRENCANAKAN' }); setAddOpen(true) }}
   325→              className={`bg-gradient-to-r ${accentColor} text-white`}>
   326→              <Plus className="w-4 h-4 mr-1" /> Tambah Program
   327→            </Button>
   328→          </div>
   329→        )}
   330→      </div>
   331→
   332→      {/* Status Stats Cards */}
   333→      <div className="grid grid-cols-4 gap-2">
   334→        {[
   335→          { key: 'DIRENCANAKAN', label: 'Direncanakan', color: 'bg-amber-50 border-amber-200 text-amber-700', count: statusCounts.DIRENCANAKAN },
   336→          { key: 'BERJALAN', label: 'Berjalan', color: 'bg-blue-50 border-blue-200 text-blue-700', count: statusCounts.BERJALAN },
   337→          { key: 'SELESAI', label: 'Selesai', color: 'bg-emerald-50 border-emerald-200 text-emerald-700', count: statusCounts.SELESAI },
   338→          { key: 'DITUNDA', label: 'Ditunda', color: 'bg-red-50 border-red-200 text-red-700', count: statusCounts.DITUNDA },
   339→        ].map(s => (
   340→          <button key={s.key} onClick={() => setFilterStatus(filterStatus === s.key ? 'ALL' : s.key)}
   341→            className={`rounded-lg border p-3 text-center transition-all ${s.color} ${filterStatus === s.key ? 'ring-2 ring-offset-1 ring-current scale-105' : 'hover:scale-105'}`}>
   342→            <div className="text-2xl font-bold">{s.count}</div>
   343→            <div className="text-xs font-medium">{s.label}</div>
   344→          </button>
   345→        ))}
   346→      </div>
   347→
   348→      {/* Search + Filter */}
   349→      <div className="flex gap-2 flex-wrap">
   350→        <Input placeholder="Cari program..." value={search} onChange={(e) => setSearch(e.target.value)} className="max-w-md flex-1" />
   351→        <Select value={filterStatus} onValueChange={setFilterStatus}>
   352→          <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   353→          <SelectContent>
   354→            <SelectItem value="ALL">Semua Status</SelectItem>
   355→            <SelectItem value="DIRENCANAKAN">Direncanakan</SelectItem>
   356→            <SelectItem value="BERJALAN">Berjalan</SelectItem>
   357→            <SelectItem value="SELESAI">Selesai</SelectItem>
   358→            <SelectItem value="DITUNDA">Ditunda</SelectItem>
   359→          </SelectContent>
   360→        </Select>
   361→      </div>
   362→
   363→      {filtered.length === 0 ? (
   364→        <EmptyState icon={Icon} title={`Belum ada program ${level}`} description={`Tambah program ${level} ${territoryName} baru, atau upload PDF program kerja.`} />
   365→      ) : (
   366→        <div className="space-y-2">
   367→          {filtered.map((item: any) => {
   368→            const statusColor = item.status === 'SELESAI' ? 'bg-emerald-100 text-emerald-700 border-emerald-200'
   369→              : item.status === 'BERJALAN' ? 'bg-blue-100 text-blue-700 border-blue-200'
   370→              : item.status === 'DITUNDA' ? 'bg-red-100 text-red-700 border-red-200'
   371→              : 'bg-amber-100 text-amber-700 border-amber-200'
   372→            return (
   373→              <div key={item.id} className="group relative rounded-lg border p-4 hover:shadow-md transition-all bg-white">
   374→                <div className="flex items-start justify-between gap-3">
   375→                  <div className="flex-1 min-w-0">
   376→                    <div className="font-semibold text-sm">{item.title}</div>
   377→                    {item.description && <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{item.description}</p>}
   378→                    <div className="flex items-center gap-3 mt-2 text-xs text-muted-foreground flex-wrap">
   379→                      {item.location && <span className="flex items-center gap-1"><MapPin className="w-3 h-3" /> {item.location}</span>}
   380→                      {item.date && <span className="flex items-center gap-1"><CalendarDays className="w-3 h-3" /> {item.date}</span>}
   381→                      <Badge variant="outline" className={`text-[11px] ${statusColor}`}>{item.status || 'DIRENCANAKAN'}</Badge>
   382→                    </div>
   383→                    {/* Action buttons row — separate from status info */}
   384→                    <div className="flex items-center gap-2 mt-2 flex-wrap">
